//===== rAthena Script ======================================
//= Personal Settings Hub
//===== By: ===================================================
//= Custom
//===== Description: =========================================
//= Central @settings menu that ties together:
//=   @spb           (Show Party Buffs)
//=   @battleinfo    (Battle Information)
//=   @packetfilter  (Packet Filter)
//=   @hideheadgear  (Hide Headgear - player controlled, any time, any map)
//= and gives quick access to @acstart (Auto Combat) and
//= @abstart (Auto Buff).
//=
//= Requires two new script commands (see script.inc / script_def.inc
//= snippets provided alongside this file):
//=   getspbstate()        -> returns 0/1
//=   getbattleinfostate() -> returns 0-15 bitmask
//= @packetfilter's state is read directly from the account
//= variable it already saves to (#BLOCK_), no patch needed.
//= @hideheadgear's state is read from the account variable
//= #HIDEHEADGEAR (0/1) that this script itself maintains -
//= no source edit needed for that one.
//= Auto Buff's state is read via getstatus(SC_AUTOBUFF), and
//= selecting it opens F_autobufftoggle's own Start/Stop/
//= Configuration menu (defined in npc/custom/autobuff.txt) -
//= no source edit needed for that one either.
//=
//= CHANGE FROM PREVIOUS VERSION:
//= The old "CheckWoEHat" NPC force-hid headgear only on WoE
//= castle maps, with no player choice. That NPC has been
//= removed. Headgear visibility is now 100% player-controlled
//= via @hideheadgear (or the "Hide Headgear" menu item below),
//= persists across maps/relogs, and is never forced on/off by
//= the server.
//============================================================
-	script	PersonalSettings	FAKE_NPC,{
function pf_subletters;

OnInit:
	bindatcmd "settings", strnpcinfo(3)+"::OnSettingsStart";
	bindatcmd "hideheadgear", strnpcinfo(3)+"::OnHideHeadgear";

	//===== DO NOT TOUCH BELOW ==================================
	.on$  = "^007000On^000000";
	.off$ = "^FF0000Off^000000";
	end;

OnSettingsStart:
	while (true) {
		mes "Choose which personal setting you want to change.";
		mes " ";
		mes "^0000FFSee Party Buffs^000000 : " + (getspbstate() ? .on$ : .off$);
		mes "^0000FFBattle Information^000000 : " + (getbattleinfostate() ? .on$ : .off$);
		mes "^0000FFPacket Filter^000000 : " + (#BLOCK_ ? .on$ : .off$);
		mes "^0000FFAuto Combat^000000 : " + (getstatus(SC_AUTOCOMBAT) ? .on$ : .off$);
		mes "^0000FFAuto Buff^000000 : " + (getstatus(SC_AUTOBUFF) ? .on$ : .off$);
		mes "^0000FFHide Headgear^000000 : " + (#HIDEHEADGEAR ? .on$ : .off$);
		next;

		switch (select("Battle Information Settings:Packet Filter Settings:Show Party Buffs Settings:Auto Combat:Auto Buff:Hide Headgear:Battle Stats / Character Resistance:Available FRO Commands:Close")) {
		case 1:
			callsub OnBattleInfoMenu;
			break;
		case 2:
			callsub OnPacketFilterMenu;
			break;
		case 3:
			callsub OnSPBMenu;
			break;
		case 4:
			close2;
			if (getstatus(SC_AUTOCOMBAT)) {
				switch (select("Stop Auto Combat:Reconfigure Auto Combat:Cancel")) {
				case 1:
					doevent "AutoCombat::OnAutoCombatEnd";
					message strcharinfo(0), "Auto Combat stopped.";
					break;
				case 2:
					doevent "AutoCombat::OnAutoCombatStart";
					end;
				case 3:
					break;
				}
			} else {
				doevent "AutoCombat::OnAutoCombatStart";
				end;
			}
			end;
		case 5:
			close2;
			callfunc "F_autobufftoggle";
			end;
		case 6:
			callsub OnHideHeadgearMenu;
			break;
		case 7:
			close2;
			atcommand "@battlestats";
			end;
		case 8:
			callsub OnCommandsList;
			break;
		case 9:
			close;
		}
	}

OnSPBMenu:
	while (true) {
		mes "Show Party Buffs is currently: " + (getspbstate() ? .on$ : .off$);
		mes " ";
		mes "When enabled, party members' key buffs (Blessing, Agi Up, Full Chemical Protection, Spirit, Devotion) show next to their name in the party window.";
		next;

		switch (select((getspbstate() ? "Turn Off" : "Turn On") + ":Back")) {
		case 1:
			atcommand "@spb";
			break;
		case 2:
			return;
		}
	}

OnBattleInfoMenu:
	while (true) {
		.@state = getbattleinfostate();
		mes "Battle Information Display: " + (.@state ? .on$ : .off$);
		mes " ";
		mes "Toggle which combat events show up in your chat window.";
		mes "^0000FFKill/Death^000000    : " + ((.@state & 1) ? .on$ : .off$);
		mes "^0000FFDispell^000000       : " + ((.@state & 2) ? .on$ : .off$);
		mes "^0000FFProvoke^000000       : " + ((.@state & 4) ? .on$ : .off$);
		mes "^0000FFMindbreaker^000000   : " + ((.@state & 8) ? .on$ : .off$);
		next;

		.@menu$ = ((.@state & 1) ? "[  X  ] Kill/Death" : "[     ] Kill/Death") + ":" +
			((.@state & 2) ? "[  X  ] Dispell" : "[     ] Dispell") + ":" +
			((.@state & 4) ? "[  X  ] Provoke" : "[     ] Provoke") + ":" +
			((.@state & 8) ? "[  X  ] Mindbreaker" : "[     ] Mindbreaker") + ":" +
			"Turn All Off:Back";

		switch (select(.@menu$)) {
		case 1:
			atcommand "@battleinfo " + (.@state ^ 1);
			break;
		case 2:
			atcommand "@battleinfo " + (.@state ^ 2);
			break;
		case 3:
			atcommand "@battleinfo " + (.@state ^ 4);
			break;
		case 4:
			atcommand "@battleinfo " + (.@state ^ 8);
			break;
		case 5:
			atcommand "@battleinfo 0";
			break;
		case 6:
			return;
		}
	}

OnPacketFilterMenu:
	@pf_filter[0] = #BLOCK_CHAT;
	@pf_filter[1] = #BLOCK_EMOTION;
	@pf_filter[2] = #BLOCK_ITEM;
	@pf_filter[3] = #BLOCK_ATTACK;
	@pf_filter[4] = #BLOCK_BUFF;
	@pf_filter[5] = #BLOCK_STATUS;
	@pf_filter[6] = #BLOCK_TARGET_SPELL;
	@pf_filter[7] = #BLOCK_AOE_SPELL;
	@pf_filter[8] = #BLOCK_MUSIC;
	@pf_filter[9] = #BLOCK_DIRECTION;

	while (true) {
		mes "Packet Filter is currently: " + (#BLOCK_ ? .on$ : .off$);
		mes " ";
		mes "Pick a category to choose who it's blocked from. Changes save immediately.";
		next;

		.@menu$ = (@pf_filter[0] ? "[  X  ] Chat" : "[     ] Chat") + ":" +
			(@pf_filter[1] ? "[  X  ] Emotions" : "[     ] Emotions") + ":" +
			(@pf_filter[2] ? "[  X  ] Item Use Effects" : "[     ] Item Use Effects") + ":" +
			(@pf_filter[3] ? "[  X  ] Attack Animations" : "[     ] Attack Animations") + ":" +
			(@pf_filter[4] ? "[  X  ] Buff Effects" : "[     ] Buff Effects") + ":" +
			(@pf_filter[5] ? "[  X  ] Status Icons" : "[     ] Status Icons") + ":" +
			(@pf_filter[6] ? "[  X  ] Target Skill Effects" : "[     ] Target Skill Effects") + ":" +
			(@pf_filter[7] ? "[  X  ] Ground/AoE Skill Effects" : "[     ] Ground/AoE Skill Effects") + ":" +
			(@pf_filter[8] ? "[  X  ] Music (Bard/Dancer Songs)" : "[     ] Music (Bard/Dancer Songs)") + ":" +
			(@pf_filter[9] ? "[  X  ] Direction" : "[     ] Direction") + ":" +
			"View Current Filters:Turn Off Entirely:Back";

		.@sel = select(.@menu$);
		if (.@sel >= 1 && .@sel <= 10) {
			@pf_cat = .@sel - 1;
			callsub OnPacketFilterSourceMenu;
		} else if (.@sel == 11) { // View Current Filters
			close2;
			atcommand "@packetfilter";
			deletearray @pf_filter[0], 10;
			end;
		} else if (.@sel == 12) { // Turn Off Entirely
			atcommand "@packetfilter off";
			deletearray @pf_filter[0], 10;
		} else { // Back
			deletearray @pf_filter[0], 10;
			return;
		}
	}

OnPacketFilterApply:
	.@cmd$ = "";
	if (@pf_filter[0]) .@cmd$ += "C" + pf_subletters(@pf_filter[0]) + " ";
	if (@pf_filter[1]) .@cmd$ += "E" + pf_subletters(@pf_filter[1]) + " ";
	if (@pf_filter[2]) .@cmd$ += "I" + pf_subletters(@pf_filter[2]) + " ";
	if (@pf_filter[3]) .@cmd$ += "A" + pf_subletters(@pf_filter[3]) + " ";
	if (@pf_filter[4]) .@cmd$ += "B" + pf_subletters(@pf_filter[4]) + " ";
	if (@pf_filter[5]) .@cmd$ += "S" + pf_subletters(@pf_filter[5]) + " ";
	if (@pf_filter[6]) .@cmd$ += "T" + pf_subletters(@pf_filter[6]) + " ";
	if (@pf_filter[7]) .@cmd$ += "G" + pf_subletters(@pf_filter[7]) + " ";
	if (@pf_filter[8]) .@cmd$ += "M" + pf_subletters(@pf_filter[8]) + " ";
	if (@pf_filter[9]) .@cmd$ += "D" + pf_subletters(@pf_filter[9]) + " ";

	if (.@cmd$ == "")
		atcommand "@packetfilter off";
	else
		atcommand "@packetfilter " + .@cmd$;
	return;

OnPacketFilterSourceMenu:
	// Self is stripped server-side for these types (see packetfilter's set_flag warnings)
	.@self_ok = (@pf_cat == 4 || @pf_cat == 7 || @pf_cat == 8);

	while (true) {
		.@v = @pf_filter[@pf_cat];
		.@menu$ = (.@self_ok ? ((.@v & 1) ? "[  X  ] Self" : "[     ] Self") : "[--] Self (not supported for this type)") + ":" +
			((.@v & 2) ? "[  X  ] Party" : "[     ] Party") + ":" +
			((.@v & 4) ? "[  X  ] Guild" : "[     ] Guild") + ":" +
			((.@v & 8) ? "[  X  ] Battleground" : "[     ] Battleground") + ":" +
			((.@v & 16) ? "[  X  ] Clan" : "[     ] Clan") + ":" +
			((.@v & 32) ? "[  X  ] Other Players" : "[     ] Other Players") + ":" +
			((.@v & 64) ? "[  X  ] Homun/Pet/Merc/Elemental" : "[     ] Homun/Pet/Merc/Elemental") + ":" +
			((.@v & 128) ? "[  X  ] Monsters" : "[     ] Monsters") + ":" +
			"Back";
		next;

		.@t = select(.@menu$);
		if (.@t == 9)
			return;
		if (.@t == 1 && !.@self_ok) {
			message strcharinfo(0), "This filter type can't block your own actions - pick another source.";
			continue;
		}
		@pf_filter[@pf_cat] ^= (1 << (.@t - 1));
		callsub OnPacketFilterApply;
	}

OnHideHeadgearMenu:
	while (true) {
		mes "Hide Headgear is currently: " + (#HIDEHEADGEAR ? .on$ : .off$);
		mes " ";
		mes "When enabled, your Top/Mid/Lower headgear sprites are hidden on every map (WoE castles included), until you switch it back off yourself. No one can force this on or off for you.";
		mes "You can also toggle it instantly anytime with ^0000FF@hideheadgear^000000 - no menu needed.";
		next;

		switch (select((#HIDEHEADGEAR ? "Show Headgear" : "Hide Headgear") + ":Back")) {
		case 1:
			callsub OnHideHeadgearToggle;
			break;
		case 2:
			return;
		}
	}

// ---- Shared toggle logic (no atcommand/end - safe to callsub from
// ---- anywhere in this same NPC, including from inside OnSettingsStart) --
OnHideHeadgearToggle:
	#HIDEHEADGEAR = !#HIDEHEADGEAR;
	callsub OnApplyHeadgearVis;
	return;

// ---- @hideheadgear command handler (bound via bindatcmd) -----------
// NOTE: do not call this via "atcommand "@hideheadgear";" from inside
// this same NPC - it will silently fail to fire a second instance
// while a dialog on this NPC is already open. Use OnHideHeadgearToggle
// via callsub instead (see OnHideHeadgearMenu above).
OnHideHeadgear:
	callsub OnHideHeadgearToggle;
	message strcharinfo(0), (#HIDEHEADGEAR ? "Headgear hidden. Use @hideheadgear again to show it." : "Headgear is now visible.");
	end;

// ---- Shared logic: paints the current preference onto the client --
OnApplyHeadgearVis:
	if (#HIDEHEADGEAR) {
		changelook LOOK_HEAD_TOP,0;
		changelook LOOK_HEAD_MID,0;
		changelook LOOK_HEAD_BOTTOM,0;
	} else {
		changelook LOOK_HEAD_TOP,getlook(LOOK_HEAD_TOP);
		changelook LOOK_HEAD_MID,getlook(LOOK_HEAD_MID);
		changelook LOOK_HEAD_BOTTOM,getlook(LOOK_HEAD_BOTTOM);
	}
	return;

// Re-apply the player's own preference on login and every map change,
// instead of the old hard-coded "hide on WoE castle maps" behavior.
OnPCLoginEvent:
	callsub OnApplyHeadgearVis;
	end;

OnPCLoadMapEvent:
	callsub OnApplyHeadgearVis;
	end;

OnCommandsList:
	.@line_count = 26; // title + blank + section header + 23 commands
	mes "[ Available Commands ]";
	setdialogalign(DIALOG_ALIGN_CENTER);
	setdialogsize(500, callfunc("CalcDialogHeight", .@line_count));
	mes " ";
	mes "^FF8000Player / General^000000";
	mes "^0000FF@settings^000000 - Open personal settings, filters, and utility shortcuts.";
	mes "^0000FF@hideheadgear^000000 - Toggle whether your headgear sprites are shown or hidden.";
	mes "^0000FF@rates^000000 - Show server rates.";
	mes "^0000FF@vip^000000 - Show VIP status, bonuses, and VIP command access.";
	mes "^0000FF@resist^000000 - Show element, damage-type, and status resistances.";
	mes "^0000FF@battlestats^000000 - Show your current battle status (ATK, MATK, DEF, etc).";
	mes "^0000FF@monsterinfo <monster>^000000 | ^0000FF@mobinfo <monster>^000000 - Show monster data and drops.";
	mes "^0000FF@whodrops <item/card>^000000 - Show monsters with the best drop rates for an item.";
	mes "^0000FF@iteminfo <item>^000000 - Show item information.";
	mes "^0000FF@whereis <monster>^000000 - Locate a monster map.";
	mes "^0000FF@time^000000 - Show server time.";
	mes "^0000FF@uptime^000000 - Show server uptime.";
	mes "^0000FF@showexp^000000 - Toggle EXP gain display.";
	mes "^0000FF@showzeny^000000 - Toggle zeny gain display.";
	mes "^0000FF@refresh^000000 - Refresh your client view.";
	mes "^0000FF@go <town>^000000 - Warp to an allowed town.";
	mes "^0000FF@storage^000000 - Open storage.";
	mes "^0000FF@alootsave^000000 - Allows you to save a specific list of items you want to pick up automatically.";
	mes "^0000FF@whosell <item>^000000 - Search vending shops.";
	mes "^0000FF@whobuy <item>^000000 - Search buying stores.";
	mes "^0000FF@autoloot <rate>^000000 - Set autoloot percentage.";
	close;

function	pf_subletters	{
	.@v = getarg(0);
	.@s$ = "";
	if (.@v & 1)   .@s$ += "S";
	if (.@v & 2)   .@s$ += "P";
	if (.@v & 4)   .@s$ += "G";
	if (.@v & 8)   .@s$ += "B";
	if (.@v & 16)  .@s$ += "C";
	if (.@v & 32)  .@s$ += "O";
	if (.@v & 64)  .@s$ += "H";
	if (.@v & 128) .@s$ += "M";
	return .@s$;
}
}
