function	script	CalcDialogHeight	{
		.@lines = getarg(0);
		.@DLG_BASE = 200;	// title bar + button row + padding
		.@DLG_LINE_H = 26;	// approx px per mes line at default font
		.@DLG_MIN = 200;	// never go smaller than this
		.@DLG_MAX = 750;	// FIX: previous cap of 600 was computed WITHOUT
					// counting the preamble (item icon/link + Server
					// Count line) that DisplayOwnersPaginated merges
					// onto page 1. Real worst case is
					// preamble_count(2) + 4 + rows(10) = 16 lines
					// => 200+16*26=616, so 600 was clamping every
					// full-page result (e.g. Fate Coin) ~16px short.
					// 750 covers that plus category searches with a
					// few "item not found" warning lines added to
					// the preamble, and is still a safety net only -
					// normal single-item results stay well under it.

		.@h = .@DLG_BASE + (.@lines * .@DLG_LINE_H);
		if (.@h < .@DLG_MIN) .@h = .@DLG_MIN;
		if (.@h > .@DLG_MAX) .@h = .@DLG_MAX;
		return .@h;
	}

	function	script	PrintHeader	{
		.@subtitle$ = getarg(0, "");
		.@n = 0;
		if (.@subtitle$ != "") {
			mes "^0080FF" + .@subtitle$ + "^000000"; .@n++;
		}
		return .@n;
	}

	function	script	SearchItem	{
		.@item_id = getarg(0);
		.@count = 0;
		.@tmp = 0;

		query_sql("SELECT IFNULL(SUM(amount),0) FROM inventory WHERE nameid="+.@item_id, .@tmp); .@count += .@tmp;
		query_sql("SELECT IFNULL(SUM(amount),0) FROM cart_inventory WHERE nameid="+.@item_id, .@tmp); .@count += .@tmp;
		query_sql("SELECT IFNULL(SUM(amount),0) FROM storage WHERE nameid="+.@item_id, .@tmp); .@count += .@tmp;
		query_sql("SELECT IFNULL(SUM(amount),0) FROM guild_storage WHERE nameid="+.@item_id, .@tmp); .@count += .@tmp;

		// ---- RODEX/MAIL (unclaimed attachments) ----
		query_sql("SELECT IFNULL(SUM(ma.amount),0) FROM mail_attachments ma WHERE ma.nameid="+.@item_id, .@tmp); .@count += .@tmp;

		if (getiteminfo(.@item_id, ITEMINFO_TYPE) == IT_CARD) {
			query_sql("SELECT IFNULL(SUM((card0="+.@item_id+")+(card1="+.@item_id+")+(card2="+.@item_id+")+(card3="+.@item_id+")),0) FROM inventory", .@tmp); .@count += .@tmp;
			query_sql("SELECT IFNULL(SUM((card0="+.@item_id+")+(card1="+.@item_id+")+(card2="+.@item_id+")+(card3="+.@item_id+")),0) FROM cart_inventory", .@tmp); .@count += .@tmp;
			query_sql("SELECT IFNULL(SUM((card0="+.@item_id+")+(card1="+.@item_id+")+(card2="+.@item_id+")+(card3="+.@item_id+")),0) FROM storage", .@tmp); .@count += .@tmp;
			query_sql("SELECT IFNULL(SUM((card0="+.@item_id+")+(card1="+.@item_id+")+(card2="+.@item_id+")+(card3="+.@item_id+")),0) FROM guild_storage", .@tmp); .@count += .@tmp;
			query_sql("SELECT IFNULL(SUM((ma.card0="+.@item_id+")+(ma.card1="+.@item_id+")+(ma.card2="+.@item_id+")+(ma.card3="+.@item_id+")),0) FROM mail_attachments ma", .@tmp); .@count += .@tmp;
		}

		return .@count;
	}

		// (the account's lowest char_id, so this still works even if that
		// account's char slot 0 was deleted).
		.@size = query_sql("SELECT c.name, s.amount FROM storage s INNER JOIN (SELECT account_id, MIN(char_id) AS char_id FROM `char` GROUP BY account_id) mc ON s.account_id = mc.account_id INNER JOIN `char` c ON c.char_id = mc.char_id WHERE s.nameid="+.@item_id, .@names$, .@amounts);

				$@card_itemnames$[.@index] = .@item_name$;
				$@card_itemids[.@index] = .@item_id;

			}

			.@size = query_sql("SELECT c.name, (ci.card0="+.@item_id+")+(ci.card1="+.@item_id+")+(ci.card2="+.@item_id+")+(ci.card3="+.@item_id+") as total FROM cart_inventory ci INNER JOIN `char` c ON ci.char_id = c.char_id WHERE ci.card0="+.@item_id+" OR ci.card1="+.@item_id+" OR ci.card2="+.@item_id+" OR ci.card3="+.@item_id, .@names$, .@amounts);

				$@card_itemids[.@index] = .@item_id;
				.@index++;
			}

			// Same account-vs-character fix as the plain storage query above.
			.@size = query_sql("SELECT c.name, (s.card0="+.@item_id+")+(s.card1="+.@item_id+")+(s.card2="+.@item_id+")+(s.card3="+.@item_id+") as total FROM storage s INNER JOIN (SELECT account_id, MIN(char_id) AS char_id FROM `char` GROUP BY account_id) mc ON s.account_id = mc.account_id INNER JOIN `char` c ON c.char_id = mc.char_id WHERE s.card0="+.@item_id+" OR s.card1="+.@item_id+" OR s.card2="+.@item_id+" OR s.card3="+.@item_id, .@names$, .@amounts);

				$@card_itemnames$[.@index] = .@item_name$;
				$@card_itemids[.@index] = .@item_id;

			}

	// Paginated owner display (10 per page by default).
	// Optional preamble (e.g. "Total Count: X") is shown ONLY on page 1,
	// merged onto the same screen as the first page of owner rows - set
	// $@dop_preamble$[] and pass the count as arg 2 before calling.
	//
	// setdialogalign/setdialogsize are ALWAYS called BEFORE any mes lines
	// are written for a given screen, on every page (including page 1,
	// where owners are actually found) - the client needs the box
	// dimensions declared up front, so this is done first on every loop
	// iteration below rather than only in the "no owners" fallback path.
	function	script	DisplayOwnersPaginated	{
		.@owner_count = getarg(0);
		.@page_size = getarg(1, 10);
		.@preamble_count = getarg(2, 0);
		.@total_pages = (.@owner_count - 1) / .@page_size + 1;
		.@page = 0;

		while(1) {
			.@start = .@page * .@page_size;
			.@end = .@start + .@page_size;
			if (.@end > .@owner_count)
				.@end = .@owner_count;
			.@rows = .@end - .@start;

			// header(1) + blank(1) + owner rows + nav(1), plus preamble on page 0
			// (+1 extra blank line on page 0 to separate the title/name line
			// from the count line(s) that follow it - see FIX NOTE 6)
			.@page_lines = 2 + .@rows + 1;
			if (.@page == 0)
				.@page_lines += .@preamble_count + 1;

			// -- setdialogsize is called here, first, on EVERY page --
			clear;
			setdialogalign(DIALOG_ALIGN_CENTER);
			setdialogsize(500, callfunc("CalcDialogHeight", .@page_lines));

			if (.@page == 0 && .@preamble_count > 0) {
				mes $@dop_preamble$[0];
				mes " ";
				for(.@k = 1; .@k < .@preamble_count; .@k++)
					mes $@dop_preamble$[.@k];
				mes " ";
				mes "^888888------------------------------^000000";
			} else {
				mes "^0080FFOwners^000000 ( Page " + (.@page + 1) + " / " + .@total_pages + " )";
			}
			mes " ";

			for(.@i = .@start; .@i < .@end; .@i++) {
				mes mesitemicon($@card_itemids[.@i]) + " [^008000" + $@card_amounts[.@i] + "^000000] " + itemlink($@card_itemids[.@i]) + " - ^0000FF" + $@card_owners$[.@i] + "^000000 ^888888(" + $@card_locations$[.@i] + ")^000000";
			}

			.@nav$ = "";
			.@n = 0;
			.@opt_next = 0;
			.@opt_prev = 0;
			if (.@page < .@total_pages - 1) { .@nav$ += "Next Page:"; .@n++; .@opt_next = .@n; }
			if (.@page > 0)                 { .@nav$ += "Previous Page:"; .@n++; .@opt_prev = .@n; }
			.@nav$ += "Close"; .@n++; .@opt_close = .@n;

			next;
			.@choice = select(.@nav$);

			if (.@choice == .@opt_next) {
				.@page++;
			}
			else if (.@choice == .@opt_prev) {
				.@page--;
			}
			else
				break;
		}

		return;
	}

	// Paginated Zeny ranking display (10 per page by default).
	// $@zeny_names$[] / $@zeny_amounts[] must be populated before calling.
	// Same rule as above: setdialogsize is called first, before any mes.
	function	script	FormatNumber	{
		.@num = getarg(0);
		.@neg = false;
		if (.@num < 0) {
			.@neg = true;
			.@num = .@num * -1;
		}
		.@str$ = .@num + "";
		.@len = getstrlen(.@str$);
		.@out$ = "";
		.@cnt = 0;
		for(.@i = .@len - 1; .@i >= 0; .@i--) {
			.@out$ = charat(.@str$, .@i) + .@out$;
			.@cnt++;
			if (.@cnt % 3 == 0 && .@i > 0)
				.@out$ = "," + .@out$;
		}
		if (.@neg)
			.@out$ = "-" + .@out$;
		return .@out$;
	}

	function	script	ShowZenyRanking	{
		.@limit = getarg(0, 50); // top 50 richest characters

		.@size = query_sql("SELECT name, zeny FROM `char` WHERE zeny > 0 ORDER BY zeny DESC LIMIT "+.@limit, $@zeny_names$, $@zeny_amounts);

		if (.@size == 0) {
			clear;
			setdialogalign(DIALOG_ALIGN_CENTER);
			setdialogsize(500, callfunc("CalcDialogHeight", 2));
			mes " ";
			mes "^888888No zeny data found.^000000";
			next;
			close2;
			return;
		}

		callfunc("DisplayZenyRankingPaginated", .@size, 10);
		return;
	}

	function	script	ShowItemResultSQL	{
		.@item_id = getarg(0);
		.@count = callfunc("SearchItem", .@item_id);

		cleararray $@card_owners$[0], "", 1024;
		cleararray $@card_amounts[0], 0, 1024;
		cleararray $@card_locations$[0], "", 1024;
		cleararray $@card_itemnames$[0], "", 1024;
		cleararray $@card_itemids[0], 0, 1024;

		.@owner_index = 0;
		if (.@count > 0)
			.@owner_index = callfunc("GetCardOwnersSQL", .@item_id, 0);

		.@pi = 0;
		$@dop_preamble$[.@pi] = mesitemicon(.@item_id) + " " + itemlink(.@item_id); .@pi++;
		if (.@count > 0) {
			$@dop_preamble$[.@pi] = "^008000Server Count: " + .@count + "^000000"; .@pi++;
		} else {
			$@dop_preamble$[.@pi] = "^FF0000Server Count: 0^000000"; .@pi++;
		}

		if (.@owner_index > 0) {
			callfunc("DisplayOwnersPaginated", .@owner_index, 10, .@pi);
		} else {
			// pad the "nobody has this" screen with an explicit note
			// (instead of just 2-3 bare lines) so it doesn't render as a
			// near-empty box next to the fuller paginated screens.
			$@dop_preamble$[.@pi] = "^888888No current owners found on the server.^000000"; .@pi++;
			clear;
			setdialogalign(DIALOG_ALIGN_CENTER);
			setdialogsize(500, callfunc("CalcDialogHeight", .@pi + 1));
			mes $@dop_preamble$[0];
			mes " ";
			for(.@j = 1; .@j < .@pi; .@j++)
				mes $@dop_preamble$[.@j];
			next;
		}

		return;
	}

			if (.@item_name$ == "null" || .@item_name$ == "") {
				.@warn_text$[.@warn_count] = "^FF0000Warning: Item ID " + .@item_id + " does not exist, skipping.^000000";
				.@warn_count++;
				continue;
			}

			.@item_total = callfunc("SearchItem", .@item_id);
			.@total += .@item_total;

			if (.@item_total > 0)
				.@owner_index = callfunc("GetCardOwnersSQL", .@item_id, .@owner_index);
		}

		// Preamble merged onto page 1 with the owner list: category name,
		// any "item not found" warnings, then the total count.
		.@pi = 0;
		$@dop_preamble$[.@pi] = "^0080FF" + .@cat_name$ + "^000000"; .@pi++;
		for(.@j = 0; .@j < .@warn_count; .@j++) {
			$@dop_preamble$[.@pi] = .@warn_text$[.@j]; .@pi++;
		}
		$@dop_preamble$[.@pi] = "^008000Total Count: " + .@total + "^000000"; .@pi++;

		if (.@owner_index > 0) {
			callfunc("DisplayOwnersPaginated", .@owner_index, 10, .@pi);
		} else {
			// same padding as ShowItemResultSQL - avoid a near-empty box.
			$@dop_preamble$[.@pi] = "^888888No current owners found on the server.^000000"; .@pi++;
			clear;
			setdialogalign(DIALOG_ALIGN_CENTER);
			setdialogsize(500, callfunc("CalcDialogHeight", .@pi + 1));
			mes $@dop_preamble$[0];
			mes " ";
			for(.@j = 1; .@j < .@pi; .@j++)
				mes $@dop_preamble$[.@j];
			next;
		}

		return;

	// Item ID only - free-text name searching has been removed. Enter a
	// numeric Item ID and the item's owners are looked up directly.
	S_SearchByItemID:
		clear;
		setdialogalign(DIALOG_ALIGN_CENTER);
		setdialogsize(500, callfunc("CalcDialogHeight", 2));
		mes " ";
		mes "Enter an Item ID:";
		next;
		input .@search$;

		if (.@search$ == "") {
			close2;
			return;
		}

		.@is_id = true;
		.@len = getstrlen(.@search$);
		for(.@k = 0; .@k < .@len; .@k++) {
			.@c$ = charat(.@search$, .@k);
			if (.@c$!="0" && .@c$!="1" && .@c$!="2" && .@c$!="3" && .@c$!="4" && .@c$!="5" && .@c$!="6" && .@c$!="7" && .@c$!="8" && .@c$!="9") {
				.@is_id = false;
				break;
			}
		}

		if (!.@is_id) {
			clear;
			setdialogalign(DIALOG_ALIGN_CENTER);
			setdialogsize(500, callfunc("CalcDialogHeight", 2));
			mes " ";
			mes "^FF0000Error: Please enter a numeric Item ID only.^000000";
			next;
			close2;
			return;
		}

		.@item_id = atoi(.@search$);
		.@item_name$ = getitemname(.@item_id);
		if (.@item_name$ == "null" || .@item_name$ == "") {
			clear;
			setdialogalign(DIALOG_ALIGN_CENTER);
			setdialogsize(500, callfunc("CalcDialogHeight", 2));
			mes " ";
			mes "^FF0000Error: Item ID " + .@item_id + " does not exist!^000000";
			next;
			close2;
			return;
		}

		callfunc("ShowItemResultSQL", .@item_id);
		return;

	OnInit:
		.npcname$ = "[^FF0000 " + strnpcinfo(1) + " ^000000]";

		setarray .cat_name$[0], "Fate Coin", "Rare MVP Cards", "Gold";
		setarray .cat_ids$[0], "30000", "4430,4408,4145", "969";
		.cat_count = 3;	// only 3 categories are defined above - keep this in
				// sync with the arrays or a blank, selectable menu
				// entry appears between "Gold" and "Search by Item ID".

		initnpctimer;
		end;

	OnTimer500:
		specialeffect EF_LJOSALFAR;
		showscript "Transparency Checker";
		initnpctimer;
		end;
	}
