function	script	F_CommaPlural	{
    .@n = getarg(0);
    .@str$ = getarg(1);
    return sprintf(getarg(2, "%s %s"), F_InsertComma(.@n), (.@n != 1 ? F_GetPlural(.@str$) : .@str$));
}

function	script	F_RODEX_SendMail_Array	{
    .@cid = getarg(0);
    .@sender$ = getarg(1, "Ragnafinest");
    .@title$ = getarg(2, "System Announcement");
    .@msg$ = getarg(3, "Hello World!\r\n\r\n- Ragnafinest");
    .@zeny = getarg(4, 0);
    .@size = getarraysize(getarg(5));
    .@qty = .@size/2;
    if (.@qty > 0) {
        freeloop(1);
        do {
            .@itm_1 = getelementofarray(getarg(5), .@i);
            .@amt_1 = getelementofarray(getarg(5), .@i+1);
            .@itm_2 = getelementofarray(getarg(5), .@i+2);
            .@amt_2 = getelementofarray(getarg(5), .@i+3);
            .@itm_3 = getelementofarray(getarg(5), .@i+4);
            .@amt_3 = getelementofarray(getarg(5), .@i+5);
            .@itm_4 = getelementofarray(getarg(5), .@i+6);
            .@amt_4 = getelementofarray(getarg(5), .@i+7);
            .@itm_5 = getelementofarray(getarg(5), .@i+8);
            .@amt_5 = getelementofarray(getarg(5), .@i+9);
            switch (.@qty) {
                case 1:
                    setarray .@itm_r[0], .@itm_1;
                    setarray .@amt_r[0], .@amt_1;
                    mail(.@cid, .@sender$, .@title$, .@msg$, .@zeny, .@itm_r, .@amt_r);
                    break;
                case 2:
                    setarray .@itm_r[0], .@itm_1, .@itm_2;
                    setarray .@amt_r[0], .@amt_1, .@amt_2;
                    mail(.@cid, .@sender$, .@title$, .@msg$, .@zeny, .@itm_r, .@amt_r);
                    break;
                case 3:
                    setarray .@itm_r[0], .@itm_1, .@itm_2, .@itm_3;
                    setarray .@amt_r[0], .@amt_1, .@amt_2, .@amt_3;
                    mail(.@cid, .@sender$, .@title$, .@msg$, .@zeny, .@itm_r, .@amt_r);
                    break;
                case 4:
                    setarray .@itm_r[0], .@itm_1, .@itm_2, .@itm_3, .@itm_4;
                    setarray .@amt_r[0], .@amt_1, .@amt_2, .@amt_3, .@amt_4;
                    mail(.@cid, .@sender$, .@title$, .@msg$, .@zeny, .@itm_r, .@amt_r);
                    break;
                default:
                    setarray .@itm_r[0], .@itm_1, .@itm_2, .@itm_3, .@itm_4, .@itm_5;
                    setarray .@amt_r[0], .@amt_1, .@amt_2, .@amt_3, .@amt_4, .@amt_5;
                    mail(.@cid, .@sender$, .@title$, .@msg$, .@zeny, .@itm_r, .@amt_r);
                    break;
            }
            .@qty -= 5;
            .@i += 10;
        } while (.@qty > 0);
        freeloop(0);
    } else {
        freeloop(1);
        while (.@zeny > 0) {
            mail(.@cid, .@sender$, .@title$, .@msg$, .@zeny > MAX_ZENY ? MAX_ZENY : .@zeny);
            .@zeny -= MAX_ZENY;
        }
        freeloop(0);
    }
    return;
}

//=========================================================================
// F_TimeShow(<time>);
//=========================================================================
function	script	F_TimeShow	{
    .@type = getarg(1,0);
    .@sec = getarg(0);
    .@day = .@sec / 86400;
    .@sec -= (.@day * 86400);
    .@hr = .@sec / 3600;
    .@sec -= (.@hr * 3600);
    .@min = .@sec / 60;
    .@sec -= (.@min * 60);

    .@str$ = "";

    .@day$ = .@type ? sprintf("%dd", .@day) : F_InsertPlural(.@day, "day", 0);
    .@hr$  = .@type ? sprintf("%dh", .@hr) : F_InsertPlural(.@hr, "hour", 0);
    .@min$ = .@type ? sprintf("%dmn", .@min) : F_InsertPlural(.@min, "minute", 0);
    .@sec$ = .@type ? sprintf("%ds", .@sec) : F_InsertPlural(.@sec, "second", 0);

    .@str$ += (.@day > 0) ? .@day$ + "" : "";
    .@str$ += (.@day > 0 && (.@hr > 0 || .@min > 0 || .@sec > 0)) ? ", " : "";
    .@str$ += (.@hr > 0) ? .@hr$ + "" : "";
    .@str$ += (.@hr > 0 && (.@min > 0 || .@sec > 0)) ? ", " : "";
    .@str$ += (.@min > 0) ? .@min$ + "" : "";
    .@str$ += (.@min > 0 && .@sec > 0) ? ", " : "";
    .@str$ += (.@sec > 0) ? .@sec$ : "";

    if (charat(.@str$, 0) == " ")
        delchar(.@str$, 0);

    if (charat(.@str$, getstrlen(.@str$) - 1) == " ")
        delchar(.@str$, 0);

    return .@str$;
}

//=========================================================================
// Display received in-game currency.
//=========================================================================
function	script	F_ShowCurrency	{
    .@var$ = getarg(0);
    .@name$ = getarg(1);
    .@point$ = .@name$ + " Point";
    setd(.@var$, getd(.@var$) + getarg(2));
    .@str$ = sprintf("[%s] You currently have %s.", .@name$ + " System", F_CommaPlural(getd(.@var$), .@point$));
    dispbottom(.@str$, 0x00FFFF);
    return;
}

function	script	F_ShowCurrency_Mes	{
    .@str$ = F_CommaPlural(getarg(0), getarg(1), " > ^777777%s ^3366FF%s^000000");
    return .@str$;
}

function	script	F_PointGain	{
    .@type = getarg(0, 0);
    .@pt = getarg(1, 0);
    .@evt$ = getarg(2);
    if (.@pt) {
        if (!.@type)
            .@str$ = sprintf("You won %s from %s.", F_CommaPlural(.@pt, "Event Point"), .@evt$);
        else
            .@str$ = sprintf("You received %s as a consolation from %s.", F_CommaPlural(.@pt, "Event Point"), .@evt$);
        message(strcharinfo(0), .@str$);
    }
    return;
}

// F_Limit(<mission_id>, <slot_idx>, <min_lv>, <duration>{, <job_1>, ...});
function	F_Limit	{
	if ((MISSION_MASK&(1 << (getarg(0) - 1))))
		return 2;
	if (getarraysize(MISSION_LIST) > 0) {
		for (.@i=0; .@i<getarraysize(MISSION_LIST); .@i++) {
			if (MISSION_LIST[.@i] == getarg(0))
				return 1;
		}
	}
	for (.@i=0; .@i<getarraysize(MISSION_DROP); .@i++) {
		if (MISSION_DROP[.@i] == getarg(0))
			return 4;
	}
	if (BaseLevel < getarg(2)) return 3;
	if (getargcount() > 4) {
		if (!getarg(4)) return 0;
		.@found = false;
		for (.@i=4; .@i<getargcount(); .@i++) {
			if (BaseClass == getarg(.@i) || BaseJob == getarg(.@i)) {
				.@found = true;
				break;
			}
		}
		if (!.@found) return 3;
		return 0;
	}
	return 0;
}

// F_MissionCheck(<mission_idx>);
function	F_MissionCheck	{
	.@mob$  = sprintf("MISSION_%d_MOBHUNT", getarg(0));
	.@kill$ = sprintf("MISSION_%d_KILL", getarg(0));
	.@itm$ = sprintf("MISSION_%d_ITMHUNT", getarg(0));

	if (getd(.@mob$)) {
		freeloop(1);
		for(.@i=0; .@i<getarraysize(getd(.@mob$)); .@i+=2) {
			.@j = .@i/2;
			if (getd(.@mob$+"["+(.@i+1)+"]") > getd(.@kill$+"["+.@j+"]")) {
				.@fail = true;
				break;
			}
		}
		freeloop(0);
	}

	if (getd(.@itm$)) {
		freeloop(1);
		for(.@i=0; .@i<getarraysize(getd(.@itm$)); .@i+=2) {
			if (getd(.@itm$+"["+(.@i+1)+"]") > countitem(getd(.@itm$+"["+.@i+"]"))) {
				.@fail = true;
				break;
			}
		}
		freeloop(0);
	}

	return (.@fail) ? 1 : 0;
}

function	F_MissionRank	{
	query_sql(sprintf("INSERT INTO `rank_mission`(`char_id`, `name`, `point`) VALUES('%d', '%s', 1) ON DUPLICATE KEY UPDATE `point` = `point` + 1", getcharid(CHAR_ID_CHAR), strcharinfo(0)));
	return;
}
