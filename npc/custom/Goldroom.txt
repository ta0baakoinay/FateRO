// ============================================================
//  Integrated Bot Check — only ever reached from OnMobKilled
//  above, which already confirmed the player is on .Map$
// ============================================================
OnBotCheckTrigger:
	if (@BOTCHECK_INPROGRESS) end;
	set @BOTCHECK_INPROGRESS, 1;

	.@name$ = strcharinfo(0);
	.@charid = getcharid(0);

	.@a = rand(1,50);
	.@b = rand(1,50);
	.@op = rand(1,2);
	if (.@op == 1) {
		.@answer = .@a + .@b;
		.@op$ = "+";
	} else {
		if (.@a < .@b) {
			.@tmp = .@a; .@a = .@b; .@b = .@tmp;
		}
		.@answer = .@a - .@b;
		.@op$ = "-";
	}

	.@attempts = 0;

BotCheckRetry:
	.@attempts++;

	mes "[^FF0000Gold Room Verification^000000]";
	mes "We've detected unusual activity in the Gold Room.";
	mes "Please solve the following to continue:";
	mes " ";
	mes "^0066FF" + .@a + " " + .@op$ + " " + .@b + " = ?^000000";
	mes " ";
	mes "You have ^FF0000" + .botcheck_timelimit + " seconds^000000 to answer.";
	mes "Attempt " + .@attempts + " / " + .botcheck_attempts + "";

	set @BOTCHECK_TIMEDOUT, 0;
	addtimer(.botcheck_timelimit * 1000, strnpcinfo(3) + "::OnBotCheckTimeout");

	input .@userinput, -99999, 99999;

	deltimer strnpcinfo(3) + "::OnBotCheckTimeout";

	if (@BOTCHECK_TIMEDOUT) {
		goto BotCheckFail;
	}

	if (.@userinput == .@answer) {
		goto BotCheckPass;
	}

	next;
	mes "[^FF0000Gold Room Verification^000000]";
	mes "That answer is incorrect.";

	if (.@attempts < .botcheck_attempts) {
		next;
		goto BotCheckRetry;
	}

BotCheckFail:
	set @BOTCHECK_INPROGRESS, 0;
	set @BOTCHECK_TIMEDOUT, 0;

	announce "[Gold Room] " + .@name$ + " failed verification and was removed as a suspected bot.", bc_all, 0xFF0000;
	logmes "[GoldRoom BotCheck] FAILED - Char: " + .@name$ + " (CID:" + .@charid + ")";

	// Removed from Gold Room entirely on failure
	warp "SavePoint", 0, 0;
	close2;

	// #BOTCHECK stays TRUE so they're re-challenged if they come back and hit another interval
	end;

BotCheckPass:
	next;
	mes "[^0000FFGold Room Verification^000000]";
	mes "Thank you, verification successful.";
	mes "You may continue farming.";

	set #BOTCHECK, 0;
	set @BOTCHECK_INPROGRESS, 0;

	// Reset kill counter so the next .botcheck_interval kills start counting fresh
	set #BOTMOBKILL_1369, 0;

	close;

OnBotCheckTimeout:
	set @BOTCHECK_TIMEDOUT, 1;
	if (attachrid(getcharid(3))) {
		dispbottom "[Gold Room] You ran out of time to respond.";
	}
	end;
}
