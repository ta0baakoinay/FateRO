	mes "Requirements:";
	disable_items;

	while(1) {

			if (@qe[0]) {

				announce strcharinfo(0)+" has failed to craft "+ItemLinkClean(.@q[0])+"!"+.@chance_text$,bc_all, 0xFF0000;

			else if ((@qe[1] & 4) || (@qe[1] & 8192)) changelook LOOK_ROBE, @qe[2];

function Add {
	if (getitemname(getarg(1)) == "null") {
		debugmes "Quest reward #"+getarg(1)+" invalid (skipped).";
		return;
	}
	setarray .@j[0],getarg(2),getarg(3),getarg(4);
	for(set .@i,5; .@i<getargcount(); set .@i,.@i+2) {
		if (getitemname(getarg(.@i)) == "null") {
			debugmes "Quest requirement #"+getarg(.@i)+" invalid (skipped).";
			return;
		} else
			setarray .@j[.@i-2],getarg(.@i),getarg(.@i+1);
	}
	copyarray getd(".q_"+getarg(0)+"_"+getarg(1)+"[0]"),.@j[0],getarraysize(.@j);
	npcshopadditem "questshop001"+("000"+getarg(0)),getarg(1),((.ShowZeny)?getarg(3):0);
	return;
}

function Chk {
	if (getarg(0) < getarg(1)) {
		set @qe[0],1;
		return "[^FF0000 X ^000000] ^FF0000";
	} else
		return "[^00AA00 O ^000000] ^00AA00";
}

function Slot {
	set .@s$,getitemname(getarg(0));
	switch(.ShowSlot) {
		case 1: if (!getitemslots(getarg(0))) return .@s$;
		case 2: if (getiteminfo(getarg(0), ITEMINFO_TYPE) == 4 || getiteminfo(getarg(0), ITEMINFO_TYPE) == 5) return .@s$+" ["+getitemslots(getarg(0))+"]";
		default: return .@s$;
	}
}

// Wraps itemlink() and strips the trailing " [0]" slot count that
// itemlink() always appends, but only when the item genuinely has
// 0 slots (so gear with real card slots still shows them, e.g. [1]).
function ItemLinkClean {
	set .@id, getarg(0);
	set .@link$, itemlink(.@id);
	if (!getitemslots(.@id)) set .@link$, replacestr(.@link$, " [0]", "");
	return .@link$;
}
}

	end;
}
