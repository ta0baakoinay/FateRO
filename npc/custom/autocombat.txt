//===== rAthena Script ======================================
//= Auto Combat
//===== By: ==================================================
//= jsn (discord: jasonch35)
//===== Modified: ============================================
//= Added daily AP timer (2 hours/day) gating Auto Combat,
//= with item-based AP top-up support. AP is tracked in
//= minutes and drains once per minute while auto combat
//= (SC_AUTOCOMBAT) is active. Resets daily per character.
//= Added AP display resync on death/fly-wing/teleport and
//= map-load so the client AP bar never goes stale after
//= SC_AUTOCOMBAT ends for a non-manual reason.
//===== Current Version: =====================================
//= 20241211 (+AP timer patch)
//= 20241212 (+AP display resync patch)
//===== Description: =========================================
//= Inspired by Ragnarok:EL
//= Duration variable is either:
//=		AC_DURATION
//=		#AC_DURATION
//= depending on AUTOCOMBAT_DURATION_CONFIG in autocombat.hpp
//= If set, AC duration is automatically calculated based on
//=		on the remaining VIP duration.
//=
//= AP TIMER NOTES:
//= - .ap_daily     = daily AP allowance in minutes (default 120 = 2 hours)
//= - .ap_ceiling   = absolute max AP can ever hold, including potion top-ups
//= - AP is tracked ACCOUNT-WIDE via #AC_AP / #AC_MaxAP / #AC_LastReset$
//=   ('#' = permanent account variable, shared by every character on
//=   the same login, per rAthena's variable scoping). This means all
//=   characters on one account draw from the same 120-minute pool —
//=   playing Auto Combat on one character spends the same AP that
//=   would be available on any other character on that account.
//= - Ap / MaxAp (no '#', native per-character stat) are kept in sync
//=   with the account pool via L_SyncApDisplay, called from every
//=   place the balance could change or the client bar could go stale:
//=   login, map load, autocombat start, every AP tick, and whenever
//=   autocombat ends for any reason (manual stop, death, fly wing,
//=   duration expiry). The account variables (#AC_AP / #AC_MaxAP)
//=   are the source of truth; Ap/MaxAp are just a mirrored display
//=   value and are never read from for gating logic.
//= - Daily reset (L_CheckApReset) compares today's date against
//=   #AC_LastReset$. It is checked on login, on map load, and on
//=   autocombat start — so even if a player never logs out, the
//=   moment the date rolls over and any of those events fires, AP
//=   is topped back up to .ap_daily (120) automatically. This is a
//=   calendar-day reset (resets at server midnight), not a rolling
//=   24h-since-last-play timer.
//= - If you ever run multiple server groups sharing one login server
//=   and want the pool to be global across all of them (not just this
//=   server group), switch '#' to '##' throughout (permanent GLOBAL
//=   account variable).
//= - The AP drain timer uses addtimer/deltimer (OnApTick), NOT
//=   initnpctimer. initnpctimer is bound to the NPC object itself
//=   (shared across all players), so it has no attached player when
//=   it later fires -> "player not attached" errors, and would also
//=   collide between players sharing this singleton NPC. addtimer
//=   captures the currently attached RID and re-attaches it
//=   automatically each time OnApTick fires, so it's safe per-player.
//=
//= PERSISTENCE ACROSS SERVER RESTARTS:
//= - #AC_AP, #AC_MaxAP, and #AC_LastReset$ are ordinary account-
//=   persistent values (same DB-backed save mechanism as character
//=   variables, just scoped to the account instead), so daily AP
//=   allowance and remaining balance are NOT lost or reset by a
//=   server restart.
//= - What does NOT survive a restart: the live SC_AUTOCOMBAT status,
//=   the addon's own mob/skill/loot selections (tracked in its
//=   compiled runtime state, not script variables), and the addtimer
//=   countdown. These are wiped along with all other server memory
//=   on restart, same as any disconnect. There's no safe way to
//=   silently resume real combat behavior after that, since we don't
//=   have access to what mobs/skills/loot rules were configured.
//= - #AC_SessionActive is a permanent account-wide flag used only to
//=   detect this case: if it's still set to 1 when any character on
//=   the account logs back in, we know their session was cut short
//=   (crash/restart) rather than ended cleanly, so we clear it and
//=   tell them to restart Auto Combat with @acstart. Their AP total
//=   is unaffected either way — it only reflects minutes actually
//=   ticked while combat was really running.
//=
//= AP DISPLAY RESYNC NOTES (why each hook exists):
//= - OnPCLoginEvent: covers normal login.
//= - OnPCLoadMapEvent: covers any map transition — fly wing, warp,
//=   GM teleport, portal, reconnect — so the bar never lags behind
//=   the real #AC_AP balance for more than one map change.
//= - OnAutoCombatEnd: fires whenever SC_AUTOCOMBAT ends for ANY
//=   reason (manual stop, death, fly wing cancelling the status,
//=   duration running out). This is the most reliable hook since
//=   it's tied directly to the status ending rather than a side
//=   effect like a map load. Only resyncs the display — does NOT
//=   run the daily reset check, since this event can fire many
//=   times per session and shouldn't re-evaluate "is it a new day"
//=   on every stop, only display the current true balance.
//============================================================
-	script	AutoCombat	FAKE_NPC,{
function skill_menu;
OnInit:
	// Max duration in minutes for autocombat in every reset
	// This is ignored if AUTOCOMBAT_DURATION_CONFIG is set to 0
	// or an item id
	.max_duration = 0;
	.max_duration_vip = 0; // set to 0 to disable

	// ---- AP timer settings ----
	.ap_daily = 120;    // daily AP allowance in minutes (2 hours)
	.ap_ceiling = 120;  // hard cap on Ap even with potion top-ups (4 hours)


	//===== DO NOT TOUCH BELOW ==================================================
	.on$ = "^007000"; .off$ = "^FF0000"; .post$ = "^000000";
	.box1$ = "[  "+.on$+"/"+.post$+"  ]  ";
	.box2$ = "[     ]  ";
	.box3$ = "[  "+.off$+"x"+.post$+"  ]  ";
	end;

OnPCLoginEvent:
	callsub L_CheckApReset;
	if (#AC_SessionActive) {
		// Session flag was still set from before we last saw this account —
		// either the server restarted mid-session or the map server crashed.
		// We do NOT try to silently resume real combat (mob/skill/loot config
		// lives in the addon's own runtime state, not in script variables),
		// so we just clear the flag and let them know to restart it manually.
		#AC_SessionActive = 0;
		message strcharinfo(0), "Your Auto Combat session was interrupted. Your AP was not affected. use @settings to resume.";
	}
	end;

OnPCLoadMapEvent:
	// Fires on every map transition (fly wing, warp, portal, GM teleport,
	// reconnect). Keeps the AP display correct even between explicit
	// login/menu events, and doubles as a daily-reset check so AP tops
	// back up the moment the date rolls over, without needing a relog.
	callsub L_CheckApReset;
	end;

OnPCDieEvent:
	// Fires on every player death, guaranteed, regardless of how the
	// core clears statuses internally on death (mass status-clear on
	// death does not always trigger the same per-status "OnEnd" hook
	// that a manual sc_end() or natural expiry does, which is why
	// OnAutoCombatEnd alone wasn't reliably catching the death case).
	// This is the direct fix for the "AP shows 0/120 after dying" bug.
	callsub L_SyncApDisplay;
	end;

L_CheckApReset:
	.@today$ = gettimestr("%Y%m%d", 20);
	if (#AC_LastReset$ != .@today$) {
		#AC_AP = .ap_daily;
		#AC_LastReset$ = .@today$;
	}
	#AC_MaxAP = .ap_daily;
	if (#AC_AP > .ap_ceiling)
		#AC_AP = .ap_ceiling;
	callsub L_SyncApDisplay;
	return;

L_SyncApDisplay:
	// Pure display sync — mirrors the account-wide AP pool onto this
	// character's native Ap/MaxAp stat/bar. Does NOT touch #AC_AP or
	// run the daily reset check, so it's safe to call as often as needed
	// (e.g. every time autocombat ends) without side effects.
	MaxAp = #AC_MaxAP;
	Ap = #AC_AP;
	return;

OnAutoCombatEnd:
	sc_end(SC_AUTOCOMBAT);
	#AC_SessionActive = 0;
	deltimer strnpcinfo(3)+"::OnApTick";
	callsub L_SyncApDisplay;
	end;

OnApTick:
	if (!getstatus(SC_AUTOCOMBAT)) {
		#AC_SessionActive = 0;
		end;
	}
	#AC_AP = #AC_AP - 1;
	MaxAp = #AC_MaxAP; // sync native ceiling every tick (fixes mid-session max_ap=0 clamp bug)
	Ap = #AC_AP;
	if (#AC_AP <= 0) {
		#AC_AP = 0;
		Ap = 0;
		sc_end(SC_AUTOCOMBAT);
		#AC_SessionActive = 0;
		message strcharinfo(0), "AP depleted for today (shared across all characters on this account). Auto Combat stopped.";
		end;
	}
	addtimer(60000, strnpcinfo(3)+"::OnApTick");
	end;

OnAutoCombatStart:
	sc_end(SC_AUTOCOMBAT);
	callsub L_CheckApReset;
	if ((.@dur = getautocombatduration(.max_duration, .max_duration_vip)) >= 0) {
		if (.@dur <= 1)
			message strcharinfo(0), "Your Auto Combat duration has expired.";
		else
			message strcharinfo(0), "You have "+callfunc("F_InsertComma", .@dur)+" seconds remaining for Auto Combat.";
	}
	message strcharinfo(0), "AP remaining: "+#AC_AP+"/"+.ap_daily+" minutes.";
	while(!.@while) {
		switch(select("Auto Combat","Attack Menu","Support Buffs","Item Loot","Teleport & Misc","Regeneration")) {
		case 1:
			if (strcharinfo(3) != mapMA$) {
				getmapxy(mapMA$, .@y, .@x, BL_PC);
				setautocombat(AC_MOBS, 0);
			}
			.@size = getavailablemobs(.@mobs);
			if (!.@size) {
				message strcharinfo(0),"No monsters in this map.";
			} else {
				message strcharinfo(0),"Select mobs to attack.";
				while(true) {
					.@select$ = "~ Start Combat ~:";
					for (.@i = 0; .@i < .@size; .@i++) {
						.@mob_name$ = getmonsterinfo(.@mobs[.@i], MOB_NAME);
						if (getautocombat(AC_MOBS, .@mobs[.@i]))
							.@select$ += .box1$ + .on$ + .@mob_name$ + .post$ + ":";
						else
							.@select$ += .box2$ + .@mob_name$ + ":";
					}
					.@select$ += "Clear Selected";
					.@s = select(.@select$) - 2;
					if (.@s == -1) {
						callsub L_CheckApReset;
						if (#AC_AP <= 0) {
							message strcharinfo(0), "You're out of AP for today (shared across all characters on this account). Use an AP item or wait for tomorrow's reset.";
							break;
						}
						getmapxy(mapMA$, .@y, .@x, BL_PC);
						if (.@dur == -1) {
							sc_start(SC_AUTOCOMBAT, 1, 1);
						} else {
							if (.@dur <= 1) {
								message strcharinfo(0), "Your Auto Combat duration has expired.";
								atcommand "@refresh";
								end;
							}
							sc_start(SC_AUTOCOMBAT, (.@dur * 1000), 0);
						}
						message strcharinfo(0), "AP remaining: "+#AC_AP+"/"+.ap_daily+" minutes.";
						deltimer strnpcinfo(3)+"::OnApTick"; // avoid stacking timers on repeat use
						#AC_SessionActive = 1;
						addtimer(60000, strnpcinfo(3)+"::OnApTick");
						.@while = 1;
						break;
					}
					if (.@s == .@size) {
						setautocombat(AC_MOBS, 0);
						continue;
					}
					setautocombat(AC_MOBS, .@mobs[.@s]);
				}
			}
			break;
		case 2:
			while(true) {
				.@menu$ = "<- Return:"+(getautocombat(AC_NORMALATK) ? .box2$ : .box1$+.on$)+"Normal Attack"+(getautocombat(AC_NORMALATK) ? "" : .post$)+":";
				.@menu$ += ""+(getautocombat(AC_RETALIATE) ? .box2$ : .box1$+.on$)+"Retaliate When Attacked"+(getautocombat(AC_RETALIATE) ? "" : .post$)+":";
				freeloop(true);
				.@size = getautocombatv(AC_ATTACKSKILL);
				for (.@i = 0; .@i < .@size; .@i++)
					.@menu$ += skill_menu(@ACINFO_ID[.@i], getautocombat(AC_ATTACKSKILL,@ACINFO_ID[.@i]));
				freeloop(false);
				.@s = select(.@menu$) - 4;
				if (.@s == -3) break;
				else if (.@s == -2) setautocombat(AC_NORMALATK);
				else if (.@s == -1) setautocombat(AC_RETALIATE);
				else {
					if (!getautocombat(AC_ATTACKSKILL,@ACINFO_ID[.@s])) {
						message(strcharinfo(0),"Please input skill level.");
						input(.@skill_lv);
						if (.@skill_lv < 1 || .@skill_lv > getskilllv(@ACINFO_ID[.@s]))
							.@skill_lv = getskilllv(@ACINFO_ID[.@s]);
					}
					setautocombat(AC_ATTACKSKILL, @ACINFO_ID[.@s], .@skill_lv);
				}
			}
			break;
		case 3:
			.@while2 = false;
			while(!.@while2) {
				switch(select("<- Return","Skill Buffs", "Item Buffs")) {
				case 1:
					.@while2 = true;
					break;
				case 2:
					while(true) {
						.@size = getautocombatv(AC_BUFFSKILL);
						.@menu$ = "<- Return:";
						for (.@i = 0; .@i < .@size; .@i++) {
							.@menu$ += skill_menu(@ACINFO_ID[.@i], getautocombat(AC_BUFFSKILL,@ACINFO_ID[.@i]));
						}
						.@s = select(.@menu$) - 2;
						if (.@s == -1) break;
						if (!getautocombat(AC_BUFFSKILL,@ACINFO_ID[.@s])) {
							message(strcharinfo(0),"Please input skill level.");
							input(.@skill_lv);
							if (.@skill_lv < 1 || .@skill_lv > getskilllv(@ACINFO_ID[.@s]))
								.@skill_lv = getskilllv(@ACINFO_ID[.@s]);
						}
						setautocombat(AC_BUFFSKILL, @ACINFO_ID[.@s], .@skill_lv);
					}
					break;
				case 3:
					disable_items;
					while(true) {
						.@size = getautocombatv(AC_BUFFITEM);
						.@menu$ = "<- Return:";
						for (.@i = 0; .@i < .@size; .@i++) {
							.@menu$ += ""+(getautocombat(AC_BUFFITEM,@ACINFO_ID[.@i]) ? .box1$+.on$ : .box2$)+""+getitemname(@ACINFO_ID[.@i])+""+(getautocombat(AC_BUFFITEM,@ACINFO_ID[.@i]) ? .post$ : "")+":";
						}
						.@s = select(.@menu$) - 2;
						if (.@s == -1) break;
						setautocombat(AC_BUFFITEM, @ACINFO_ID[.@s]);
					}
					enable_items;
					break;
				}
			}
			break;
		case 4:
			.@while2 = false;
			while(!.@while2) {
				.@menu$ = "<- Return:";
				.@menu$ += ""+(getautocombatv(AC_LOOTITEM) == AC_LOOT_NONE ? .box1$+.on$ : .box2$)+"Disable Looting"+(getautocombatv(AC_LOOTITEM) == AC_LOOT_NONE ? .post$ : "")+":";
				.@menu$ += ""+(getautocombatv(AC_LOOTITEM) == AC_LOOT_ALL ? .box1$+.on$ : .box2$)+"Loot Everything"+(getautocombatv(AC_LOOTITEM) == AC_LOOT_ALL ? .post$ : "")+":";
				.@menu$ += ""+(getautocombatv(AC_LOOTITEM) == AC_LOOT_GROUP_1 ? .box1$+.on$ : .box2$)+"Loot Group 1 Only"+(getautocombatv(AC_LOOTITEM) == AC_LOOT_GROUP_1 ? .post$ : "")+":";
				.@menu$ += ""+(getautocombatv(AC_LOOTITEM) == AC_LOOT_GROUP_2 ? .box1$+.on$ : .box2$)+"Loot Group 2 Only"+(getautocombatv(AC_LOOTITEM) == AC_LOOT_GROUP_2 ? .post$ : "")+":";
				.@s = select(.@menu$) - 3;
				if (.@s == -2) break;
				if (.@s <= 0) {
					setautocombat(AC_LOOTITEM, .@s);
					continue;
				}
				.@break = false;
				while(!.@break) {
					.@group$ = "<- Return:";
					.@group$ += ""+(getautocombatv(AC_LOOTITEM) == .@s ? .box1$+.on$ : .box2$)+" Enable Loot Group "+.@s+""+(getautocombatv(AC_LOOTITEM) == .@s ? .post$ : "")+":";
					.@group$ += "[ > ]  Add Item:";
					.@size = query_sql("SELECT `item_id` FROM `autocombat_items` WHERE `char_id` = "+getcharid(0)+" AND `type` = "+(.@s + 1)+"", .@itemid);
					for (.@i = 0; .@i < .@size; .@i++)
						.@group$ += "- " + getitemname(.@itemid[.@i]) + ":";
					.@t = select(.@group$) - 4;
					if (.@t == -3) {
						.@break = true;
					} else if (.@t == -2) {
						setautocombat(AC_LOOTITEM, .@s);
						for (.@i = 0; .@i < .@size; .@i++) {
							setautocombat(AC_LOOTITEM, .@itemid[.@i]);
						}
					} else if (.@t == -1) {
						input(.@additem);
						if (getitemname(.@additem) == "null") {
							message(strcharinfo(0), "Item ID not found.");
							break;
						}
						if (getautocombat(AC_LOOTITEM, .@additem)) {
							message(strcharinfo(0), ""+getitemname(.@additem)+" is already on group "+.@s+".");
							break;
						}
						query_sql("INSERT INTO `autocombat_items` (`char_id`, `type`, `item_id`) VALUES ("+getcharid(0)+", "+(.@s + 1)+", "+.@additem+")");
						if (getautocombatv(AC_LOOTITEM) == .@s)
							setautocombat(AC_LOOTITEM, .@additem);
					} else {
						query_sql("DELETE FROM `autocombat_items` WHERE `char_id` = "+getcharid(0)+" AND `type` = "+(.@s + 1)+" AND `item_id` = "+.@itemid[.@t]+"");
						if (getautocombatv(AC_LOOTITEM) == .@s)
							setautocombat(AC_LOOTITEM, .@itemid[.@t]);
					}
				}
			}
			break;
		case 5:
			.@break = 0;
			while(!.@break) {
				.@menu$ = "<- Return:";
				.@menu$ += ""+(getautocombat(AC_TELEPORT, 0) ? .box1$ : .box2$)+" Use Teleport Skill:";
				.@menu$ += ""+(getautocombat(AC_TELEPORT, 1) ? .box1$ : .box2$)+" Use Fly Wing:";
				.@menu$ += ""+(getautocombat(AC_TELEPORT, 2) ? .box1$ : .box2$)+" Teleport from MVP:";
				.@menu$ += ""+(getautocombat(AC_TELEPORT, 3) ? "[ "+.on$+""+getautocombat(AC_TELEPORT, 3)+"%"+.post$+" ]" : .box3$)+" Emergency Teleport HP Treshold:";
				.@menu$ += ""+(getautocombat(AC_TELEPORT, 4) ? "[ "+.on$+""+getautocombat(AC_TELEPORT, 4)+"s"+.post$+" ]" : .box3$)+" Teleport No Mob Delay:";
				.@menu$ += "[ "+.on$+""+getautocombatv(AC_ENDCONDITION)+""+.post$+" ] When AC Ends:";
				switch(select(.@menu$)) {
				case 1:
					.@break = 1;
					break;
				case 2:
					setautocombat(AC_TELEPORT, 0, 0);
					break;
				case 3:
					setautocombat(AC_TELEPORT, 1, 0);
					break;
				case 4:
					setautocombat(AC_TELEPORT, 2, 0);
					break;
				case 5:
					message strcharinfo(0),"Please input from 3 to 90 percent HP to emergency teleport.";
					input(.@minhp);
					if (.@minhp < 3 || .@minhp > 90) .@minhp = 0; // set default
					setautocombat(AC_TELEPORT, 3, .@minhp);
					break;
				case 6:
					message strcharinfo(0),"Please input up to 120 seconds to teleport if no monsters are found.";
					input(.@teleport);
					if (.@teleport < 1 || .@teleport > 120) .@teleport = 0; // set default
					setautocombat(AC_TELEPORT, 4, .@teleport);
					break;
				case 7:
					setautocombat(AC_ENDCONDITION);
					break;
				}
			}
			break;
		case 6:
			.@break = 0;
			while(!.@break) {
				.@menu$ = "<- Return:";
				.@size = getautocombatv(AC_HEALSKILL);
				for (.@i = 0; .@i < .@size; .@i++) {
					if (getskilllv(@ACINFO_ID[.@i]) > 0) {
						if (@ACINFO_VAL[.@i]) {
							.@menu$ += "[ "+.on$+"Lv "+@ACINFO_VAL[.@i]+""+.post$+" ] "+.on$+getskillname(@ACINFO_ID[.@i])+.post$+" @ "+@ACINFO_VAL2[.@i]+"% "+(.@i == 2 ? "SP" : "HP")+":";
						} else {
							.@menu$ += .box2$+getskillname(@ACINFO_ID[.@i])+":";
						}
					} else .@menu$ += ":";
				}
				.@menu$ += "HP Potions:SP Potions:";
				.@menu$ += "[  "+.on$+""+getautocombat(AC_SIT,0)+"%"+.post$+"  ] Sit to Regen HP:";
				.@menu$ += ""+(getautocombat(AC_SIT,1) ? "[  "+.on$+""+getautocombat(AC_SIT,1)+"%"+.post$+"  ]" : .box3$)+" Sit to Regen SP:";
				switch((.@s = select(.@menu$))) {
				case 1:
					.@break = 1;
					break;
				case 2:
				case 3:
				case 4:
					.@s -= 2;
					if (!getautocombat(AC_HEALSKILL, .@s)) {
						message(strcharinfo(0),"Please input skill level.");
						input(.@skill_lv);
						message strcharinfo(0),"Please input minimun "+(.@s == 2 ? "SP" : "HP")+" to use "+getskillname(@ACINFO_ID[.@s])+".";
						input(.@minhp);
						if (.@skill_lv < 1 || .@skill_lv > getskilllv(@ACINFO_ID[.@s]))
							.@skill_lv = getskilllv(@ACINFO_ID[.@s]);
						if (.@minhp < 1 || .@minhp > 99) .@minhp = 1; // set default
					} else {
						.@skill_lv = .@minhp = 0;
					}
					setautocombat(AC_HEALSKILL, .@s, .@skill_lv, .@minhp);
					break;
				case 5:
					while(true) {
						.@size = getautocombatv(AC_HPPOTION);
						.@menu$ = "<- Return:";
						for (.@i = 0; .@i < .@size; .@i++) {
							if ((.@rate = getautocombat(AC_HPPOTION, @ACINFO_ID[.@i])) > 0)
								.@menu$ += "[ "+.on$+""+.@rate+"% HP"+.post$+" ] "+.on$+getitemname(@ACINFO_ID[.@i])+.post$+":";
							else
								.@menu$ += .box2$+getitemname(@ACINFO_ID[.@i])+":";
						}
						.@s = select(.@menu$) - 2;
						if (.@s == -1) break;
						if (getautocombat(AC_HPPOTION, @ACINFO_ID[.@s])) {
							setautocombat(AC_HPPOTION, @ACINFO_ID[.@s], 0);
							continue;
						}
						message strcharinfo(0),"Please input HP threshold to use "+getitemname(@ACINFO_ID[.@s])+".";
						input(.@minhp);
						if (.@minhp < 1 || .@minhp > 99) {
							message strcharinfo(0),"Invalid HP percentage. Please try again.";
							continue;
						}
						setautocombat(AC_HPPOTION, @ACINFO_ID[.@s], .@minhp);
					}
					break;
				case 6:
					while(true) {
						.@size = getautocombatv(AC_SPPOTION);
						.@menu$ = "<- Return:";
						for (.@i = 0; .@i < .@size; .@i++) {
							if ((.@rate = getautocombat(AC_SPPOTION, @ACINFO_ID[.@i])) > 0)
								.@menu$ += "[ "+.on$+""+.@rate+"% SP"+.post$+" ] "+.on$+getitemname(@ACINFO_ID[.@i])+.post$+":";
							else
								.@menu$ += .box2$+getitemname(@ACINFO_ID[.@i])+":";
						}
						.@s = select(.@menu$) - 2;
						if (.@s == -1) break;
						if (getautocombat(AC_SPPOTION, @ACINFO_ID[.@s])) {
							setautocombat(AC_SPPOTION, @ACINFO_ID[.@s], 0);
							continue;
						}
						message strcharinfo(0),"Please input SP threshold to use "+getitemname(@ACINFO_ID[.@s])+".";
						input(.@minhp);
						if (.@minhp < 1 || .@minhp > 99) {
							message strcharinfo(0),"Invalid SP percentage. Please try again.";
							continue;
						}
						setautocombat(AC_SPPOTION, @ACINFO_ID[.@s], .@minhp);
					}
					break;
				case 7:
					message strcharinfo(0),"Please input from 5 to 75 percent HP to sit.";
					input(.@minhp);
					if (.@minhp < 5 || .@minhp > 75) .@minhp = 5; // set default
					setautocombat(AC_SIT,0,.@minhp);
					break;
				case 8:
					message strcharinfo(0),"Please input from 5 to 75 percent SP to sit. Set to 0 to disable.";
					input(.@minsp);
					if (.@minsp < 5 || .@minsp > 75) .@minsp = 0;
					setautocombat(AC_SIT,1,.@minsp);
					break;
				}
			}
		}
	}
	end;

function	skill_menu	{
	.@skill_id = getarg(0);
	.@active = getarg(1);
	if (.@active) {
		.@menu$ = "[ "+.on$+"Lv "+.@active+""+.post$+" ] "+.on$+getskillname(.@skill_id)+.post$+":";
	} else {
		.@menu$ = .box2$+getskillname(.@skill_id)+":";
	}
	return .@menu$;
}
}

function	script	F_AutoCombatItem	{
	doevent("AutoCombat::OnAutoCombatStart");
	return;
	if (getstatus(SC_AUTOCOMBAT)) {
		sc_end(SC_AUTOCOMBAT);
		return;
	}
	.@dur = getautocombatduration(getvariableofnpc(.max_duration, "AutoCombat"), getvariableofnpc(.max_duration_vip, "AutoCombat"));
	switch(.@dur) {
	case -1:
		sc_start(SC_AUTOCOMBAT, 1, 1);
		break;
	case 0:
	case 1:
		message strcharinfo(0), "Your Auto Combat duration has expired.";
		atcommand "@refresh";
		break;
	default:
		message strcharinfo(0), "You have "+callfunc("F_InsertComma", .@dur)+" seconds remaining for Auto Combat.";
		sc_start(SC_AUTOCOMBAT, (.@dur * 1000), 0);
		break;
	}
	return;
}
