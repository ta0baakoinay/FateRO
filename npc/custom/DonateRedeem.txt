//===== rAthena Script =======================================
//= Donor Rewards Redeemer (+ Donation Rankings)
//===== Description: =========================================
//= Main menu:
//=   1) Redeem unclaimed rewards - reads/updates cp_redeemlog,
//=      populated by FluxCP's stock item shop checkout
//=      (modules/purchase/checkout.php) whenever a player spends
//=      credits on an item. This is FluxCP's real, already-working
//=      pipeline - no custom FluxCP code needed for this part.
//=   2) View Donation Rankings - ranks accounts by total credits
//=      ever received via completed PayPal/IPN transactions,
//=      read directly from cp_txnlog (FluxCP's real donation
//=      ledger, populated by Flux_PaymentNotifyRequest on every
//=      successful donation). This replaces the earlier design
//=      that read from cp_donation_requests, a table nothing in
//=      FluxCP (live site or backups) ever actually wrote to.
//=
//= NOTE: ShowDonationRankings/DisplayDonationRankingsPaginated
//= call callfunc("CalcDialogHeight", ...) and
//= callfunc("FormatNumber", ...) — these already exist on this
//= server (Searcher.txt).
//=============================================================

maintown,159,182,4	script	Donor Rewards Redeemer	117,{

	.@redeemTable$ = "cp_redeemlog";
	.@maxRedeem    = 20; // max unique rewards claimed in one visit

	mes "[Donor Rewards Redeemer]";
	mes "Well met! How can I help you today?";
	next;

	switch(select("Redeem my rewards:View Donation Rankings:Never mind")) {

	case 1:
		.@account_id = getcharid(3);
		.@count = query_sql(
			"SELECT `id`,`nameid`,`quantity` FROM `" + .@redeemTable$ + "` " +
			"WHERE `account_id` = " + .@account_id + " AND `redeemed` = 0 " +
			"LIMIT " + .@maxRedeem,
			.@id, .@nameid, .@quantity
		);

		if (.@count <= 0) {
			mes "[Donor Rewards Redeemer]";
			mes "You have nothing waiting for you right now.";
			close;
		}

		mes "[Donor Rewards Redeemer]";
		mes "You have " + .@count + " reward(s) waiting! Redeeming now...";
		next;

		.@totalFateCoins = 0;

		for (.@i = 0; .@i < .@count; .@i++) {
			query_sql "UPDATE `" + .@redeemTable$ + "` SET `char_id` = " + getcharid(0) +
				", `redeemed` = 1, `redemption_date` = NOW() WHERE `id` = " + .@id[.@i];

			getitem .@nameid[.@i], .@quantity[.@i];

			// Track Fate Coin (item 30000) specifically for the announcement below.
			if (.@nameid[.@i] == 30000) {
				.@totalFateCoins += .@quantity[.@i];
			}

			mes "[Donor Rewards Redeemer]";
			mes "Delivered " + .@quantity[.@i] + "x " + getitemname(.@nameid[.@i]) + "!";
			next;
		}

		mes "[Donor Rewards Redeemer]";
		mes "Thank you for supporting the server!";

		if (.@totalFateCoins > 0) {
			announce strcharinfo(0) + " claimed a donation worth " + .@totalFateCoins + " Fate Coin(s)!", bc_all|bc_blue;
		}

		close;

	case 2:
		callfunc("ShowDonationRankings");
		close;

	case 3:
		mes "[Donor Rewards Redeemer]";
		mes "Come back anytime!";
		close;
	}
}

// =============================================================
// Donation Rankings — ranks accounts by total credits received
// from completed PayPal/IPN transactions (cp_txnlog.payment_status
// = 'Completed'), FluxCP's real donation ledger populated by
// Flux_PaymentNotifyRequest on every successful donation. One row
// per account via the MIN(char_id)-per-account join trick.
// =============================================================

function	script	ShowDonationRankings	{
	.@limit = getarg(0, 50); // top 50 donors

	.@size = query_sql(
		"SELECT IFNULL(c.name, CONCAT('Account #', tl.account_id)), SUM(tl.credits), SUM(CAST(tl.mc_gross AS DECIMAL(10,2))) " +
		"FROM cp_txnlog tl " +
		"LEFT JOIN (SELECT account_id, MIN(char_id) AS char_id FROM `char` GROUP BY account_id) mc ON mc.account_id = tl.account_id " +
		"LEFT JOIN `char` c ON c.char_id = mc.char_id " +
		"WHERE tl.payment_status = 'Completed' " +
		"GROUP BY tl.account_id " +
		"ORDER BY SUM(tl.credits) DESC " +
		"LIMIT " + .@limit,
		$@donor_names$, $@donor_coins, $@donor_php);

	if (.@size == 0) {
		clear;
		setdialogalign(DIALOG_ALIGN_CENTER);
		setdialogsize(500, callfunc("CalcDialogHeight", 2));
		mes " ";
		mes "^888888No completed donations found yet.^000000";
		next;
		return;
	}

	callfunc("DisplayDonationRankingsPaginated", .@size, 10);
	return;
}

function	script	DisplayDonationRankingsPaginated	{
	.@total = getarg(0);
	.@page_size = getarg(1, 10);
	.@total_pages = (.@total - 1) / .@page_size + 1;
	.@page = 0;

	while(1) {
		.@start = .@page * .@page_size;
		.@end = .@start + .@page_size;
		if (.@end > .@total)
			.@end = .@total;
		.@rows = .@end - .@start;

		.@page_lines = 2 + .@rows + 1;

		clear;
		setdialogalign(DIALOG_ALIGN_CENTER);
		setdialogsize(500, callfunc("CalcDialogHeight", .@page_lines));

		mes "^0080FFDonation Rankings^000000 ( Page " + (.@page + 1) + " / " + .@total_pages + " )";
		mes " ";

		for(.@i = .@start; .@i < .@end; .@i++) {
			mes "^888888#" + (.@i + 1) + "^000000 ^0000FF" + $@donor_names$[.@i] + "^000000 - ^008000" + $@donor_coins[.@i] + " Fate Coin(s)^000000 ^888888($" + callfunc("FormatNumber", $@donor_php[.@i]) + " USD donated)^000000";
		}

		.@nav$ = "";
		.@n = 0;
		.@opt_next = 0;
		.@opt_prev = 0;
		if (.@page < .@total_pages - 1) { .@nav$ += "Next Page:"; .@n++; .@opt_next = .@n; }
		if (.@page > 0)                 { .@nav$ += "Previous Page:"; .@n++; .@opt_prev = .@n; }
		.@nav$ += "Close"; .@n++; .@opt_close = .@n;

		next;
		.@choice = select(.@nav$);

		if (.@choice == .@opt_next) {
			.@page++;
		}
		else if (.@choice == .@opt_prev) {
			.@page--;
		}
		else
			break;
	}

	return;
}
