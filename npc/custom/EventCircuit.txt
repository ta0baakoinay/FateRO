// ============================================================
// Event Circuit — ONE NPC, ONE script, controls everything: the
// info board, Dice, LMS, NvZ, Poring Catcher, AND Find the GM.
// There is no second/duplicate NPC object anywhere — during Find
// GM this NPC physically unitwarps itself away from town and
// back, rather than needing a separate walking unit standing on
// the same tile (which was rendering as two overlapping sprites).
// While it's off hiding, the info board isn't in town either
// (it's the same object) — it returns when the event ends.
//
// Only one event can run at a time (.active_event$ guard) —
// Find GM now participates in that guard too, which it didn't
// before.
//
// TIMER FIX: instead of 24 separate OnClockXXXX bodies (one per
// event, out of sync with each other, and Find GM excluded from
// the rotation entirely and running off its own OnMinute00
// checker), every hour now falls into ONE shared block that
// picks the event via hour % 5 against a single label array.
// All 5 events rotate evenly through the day and never overlap.
//
// TIMER FIX #2: Find GM's old 30-minute safety cutoff
// (OnTimer1800000) was broken — initnpctimer was called again at
// the start of every round, which resets the npc timer clock to
// 0 each time, so the 30-minute counter never actually
// accumulated across rounds. Replaced with a real elapsed-time
// check using gettimetick(2) captured once when the event starts.
// ============================================================

		mes "Live status:";

		mes " ";
		// #EVENT_POINTS is an account variable stored in acc_reg_num.
		// The index is 0 for the scalar account variable.
		.@count = query_sql(
			"SELECT r.`account_id`, r.`value`, MIN(c.`name`) "
			+"FROM `acc_reg_num` r "
			+"LEFT JOIN `char` c ON c.`account_id` = r.`account_id` "
			+"WHERE r.`key` = '#EVENT_POINTS' AND r.`index` = 0 "
			+"GROUP BY r.`account_id`, r.`value` "
			+"ORDER BY r.`value` DESC, r.`account_id` ASC "
			+"LIMIT 10",
			.@rank_aid, .@rank_points, .@rank_name$
		);
		if ( !.@count ) {
			mes "No Event Points have been recorded yet.";
			close;

OnStart:
	if ( .@atcmd_numparameters < 1 ) {
		dispbottom "Usage: @eventstart <dice|lms|poring|nvz|findgm>";
		dispbottom "Example: @eventstart dice";
		end;
	}

	.@event$ = strtolower(.@atcmd_parameters$[0]);

	if ( .@event$ == "dice" ) {
		donpcevent strnpcinfo(0)+"::OnDiceStart";
		end;
	}

	if ( .@event$ == "lms" ) {
		donpcevent strnpcinfo(0)+"::OnLMSStart";
		end;
	}

	if ( .@event$ == "poring" || .@event$ == "poringcatcher" ) {
		donpcevent strnpcinfo(0)+"::OnPoringCatcherStart";
		end;
	}

	if ( .@event$ == "nvz" ) {
		donpcevent strnpcinfo(0)+"::OnNvZStart";
		end;
	}

	if ( .@event$ == "findgm" || .@event$ == "find" ) {
		donpcevent strnpcinfo(0)+"::OnFindGMEventStart";
		end;
	}

	dispbottom "Invalid event.";
	dispbottom "Usage: @eventstart <dice|lms|poring|nvz|findgm>";
	end;

OnEventStop:
	.active_event$ = "";
	.dice_start = false;
	.lms_start = false;
	.nvz_start = false;
	.poring_start = false;
	.findgm_status = 0;
	stopnpctimer;
	killmonsterall "quiz_01";
	killmonsterall "poring_w01";
	pvpoff "06guild_01";
	delwaitingroom;
	for(.@i = 1;.@i <= 4;.@i++) {
		disablenpc "Back#"+.@i;
		disablenpc "Box "+.@i;

	mapwarp "06guild_01","maintown",151,103;
	mapwarp "quiz_01","maintown",151,103;
	mapwarp "poring_w01","maintown",151,103;
	donpcevent "Dev Prime#findgm::OnFindGMForceReset";

	end;

// @join is also an atcommand binding. It MUST NOT open a dialog because
// players are allowed to use @join from anywhere. npc_scriptcont performs
// npc_checknear on dialog continuation, which is incompatible with this use.
OnJoinEvent:
	if ( strcharinfo(3) == "guild_vs5" ) {
		dispbottom "You cannot join events while in guild_vs5.";
		end;
	}

	.@start = 0;
	.@map$ = "";
	.@x = 0;
	.@y = 0;

	if ( .dice_start == 1 ) {
		.@start = .dice_start;
		.@map$ = "quiz_02"; .@x = 55; .@y = 87;
	} else if ( .lms_start == 1 ) {
		.@start = .lms_start;
		.@map$ = "06guild_01"; .@x = 0; .@y = 0;
	} else if ( .poring_start == 1 ) {
		.@start = .poring_start;
		.@map$ = "poring_w01"; .@x = 100; .@y = 100;
	} else if ( .nvz_start == 1 ) {
		if ( BaseLevel > 1 || Class != 0 ) {
			dispbottom "Only Lv 1 Novice can enter.";
			end;
		}
		getinventorylist;
		if ( @inventorylist_amount ) {
			dispbottom "You must store all your items first.";
			@inventorylist_amount = false;
			end;
		}
		sc_start SC_DECREASEAGI,600000,10;
		.@start = .nvz_start;
		.@map$ = "quiz_01"; .@x = 42; .@y = 369;
	} else if ( .findgm_status >= 1 ) {
		dispbottom "Find GM doesn't warp you in. Find Dev Prime on his current map and enter his waiting room.";
		end;
	}

	if ( !.@start ) {
		dispbottom "No event is currently accepting participants.";
		end;
	}

	sc_end SC_ALL;
	percentheal 100,100;
	warp .@map$,.@x,.@y;

	#EVENT_POINTS += 1;
	dispbottom "You received 1 Event Point for participating! You now have "+#EVENT_POINTS+" Event Points.";
	end;

// ---------------- Dice ----------------
OnDiceStart:
	if ( .active_event$ != "" ) end;
	.active_event$ = "Dice";
	.dice_start = 1;
	for(.@i = 1;.@i <= 4;.@i++) {
		disablenpc "Back#"+.@i;
		disablenpc "Box "+.@i;
	}
	announce "Dice: We are going to have a Dice event.",0;
	sleep 10000;
	if ( .active_event$ != "Dice" ) end;

	announce "Dice: For those who wants to join, use @join",0;
	sleep 10000;
	if ( .active_event$ != "Dice" ) end;

	announce "Dice: After 1 Minute Event will close.",0;
	sleep 10000;
	if ( .active_event$ != "Dice" ) end;

	announce "Dice: So please if you want to join. use @join",0;
	for(.@w = 0; .@w < 6; .@w++) {
		sleep 10000;
		if ( .active_event$ != "Dice" ) end;
	}
	announce "Dice: The entry to the event is now closed.",bc_all,0x0ceedb;
	if ( !getmapusers("quiz_02") ) {
		announce "Dice: Not enough participants.",bc_all,0x0ceedb;
		.dice_start = false;
		.active_event$ = "";
		end;
	}
	.dice_start = 2;
	sleep 3500;
	mapannounce "quiz_02","We're about to begin the event.",bc_map,0x0ceedb;
	sleep 2500;
	mapannounce "quiz_02","The rules are simple.",bc_map,0x0ceedb;
	sleep 2500;
	mapannounce "quiz_02","You'll have to enter one of the 4 avaiaible boxes.",bc_map,0x0ceedb;
	sleep 2500;
	mapannounce "quiz_02","Then a four sides dice will be thrown. Wrong boxes are thrown out immediately.",bc_map,0x0ceedb;
	sleep 2500;
	mapannounce "quiz_02","Remember to enter the boxes!",bc_map,0x0ceedb;
	sleep 2000;
	mapannounce "quiz_02","That's all. Let's begin.",bc_map,0x0ceedb;
	sleep 3000;
	while(!.@end && .dice_start && .@rounds < 5) {
		for(.@i = 1;.@i <= 4;.@i++) {
			enablenpc "Back#"+.@i;
			enablenpc "Box "+.@i;
		}
		sleep 1500;
		mapannounce "quiz_02","The dice will soon be thrown. Enter the desired box clicking the metalic number inside them.",bc_map,0x0ceedb;
		for(.@w = 0; .@w < 4; .@w++) {
			sleep 5000;
			if ( .active_event$ != "Dice" ) end;
		}
		for(.@i = 5;.@i > 0;.@i--) {
			mapannounce "quiz_02",.@i+"...",bc_map,0x0ceedb;
			sleep 990;
		}
		for(.@i = 1;.@i <= 4;.@i++) {
			disablenpc "Back#"+.@i;
			disablenpc "Box "+.@i;
		}
		areawarp "quiz_02",32,95,79,76,"maintown",151,103;
		areawarp "quiz_02",80,85,80,76,"maintown",151,103;
		areawarp "quiz_02",31,85,31,76,"maintown",151,103;
		.@n = rand(1,4);
		mapannounce "quiz_02","Time! The dice shows the number "+.@n+".",bc_map,0x0ceedb;
		setarray .@cords1[0],33,41;
		setarray .@cords2[0],45,53;
		setarray .@cords3[0],57,65;
		setarray .@cords4[0],69,77;
		for(.@i = 1;.@i <= 4;.@i++) {
			if(.@i != .@n)
				areawarp "quiz_02",getd(".@cords"+.@i+"[0]"),52,getd(".@cords"+.@i+"[1]"),73,"maintown",151,103;
		}
		sleep 3000;
		.@players = getmapusers("quiz_02");
		if(.@players <= 3 && .dice_start && .@players) {
			mapannounce "quiz_02","The event is now over! Winners, enjoy your prizes!",bc_map,0x0ceedb;
			.@size = getmapunits(BL_PC,"quiz_02",.@players);
			.dice_start = false;
			for(.@i = 0;.@i < .@size;.@i++) {
				attachrid .@players[.@i];
				announce strcharinfo(0)+" has won the Dice Event.",bc_all,0x0ceedb;
				if ( $event_options&1 ) {
					for ( .@j = 0; .@j < $event_item_arr; .@j += 2 )
						getitem $event_item_reward[.@j], $event_item_reward[.@j+1];
				}
				#EVENT_POINTS += 2;
				dispbottom "You won the event! You received 2 additional Event Points (3 total for winning). You now have "+#EVENT_POINTS+" Event Points.";
				warp "maintown",151,103;
			}
		} else if( !.@players ) {
			.@end = true;
			.dice_start = false;
			announce "There was no winner on the Dice Event.",bc_all,0x0ceedb;
		} else {
			mapannounce "quiz_02","There's no clear winners yet, let's roll the dice again.",bc_map,0x0ceedb;
		}
		.@rounds++;
	}
	if(.@rounds >= 5) {
		mapannounce "quiz_02","There were no winner clear in all of the event. The event is now over.",bc_map,0x0ceedb;
		.dice_start = false;
	}
	.active_event$ = "";
	end;

// ---------------- Find the GM ----------------
// Entry point from the hourly rotation, runs on "Event Circuit"
// itself (which stays put in town). It hands off the actual
// hide-and-seek to the separate "Dev Prime#findgm" NPC (declared
// at the bottom of this file) — that one is invisible/parked
// until now, so it never overlaps with Event Circuit in town.
OnFindGMEventStart:
	if ( .active_event$ != "" ) end;
	.active_event$ = "FindGM";
	.findgm_status = 1;
	.findgm_startTick = gettimetick(2);
	for(.@i = .findgm_countdown; .@i > 0; .@i--) {
		announce "[Find "+.npcName$+"] : The event will begin in "+callfunc("F_InsertPlural",.@i,"minute")+"!",bc_all|bc_yellow;
		for(.@w = 0; .@w < 6; .@w++) {
			sleep 10000;
			if ( .active_event$ != "FindGM" ) end;
		}
	}
	announce "[Find "+.npcName$+"] : The event has now begun!",bc_all|bc_yellow;
	donpcevent "Dev Prime#findgm::OnFindGMIntro";
	end;

// Runs on "Dev Prime#findgm" (invoked via strnpcinfo(0) from within
// its own OnFindGMRoundStart/OnFindGMRoundEnd below), so "self"
// here is the walking unit, not the town board.
OnFindGMEnd:
	.findgm_status = 0;
	.active_event$ = "";
	announce "[Find "+.npcName$+"]: The event is over! thank you for participating.",bc_all|bc_yellow;
	delwaitingroom;
	unitwarp getnpcid(0),"maintown",114,156;
	hideonnpc strnpcinfo(0);
	end;

// Also runs on "Dev Prime#findgm" — makes it visible and sends it
// off to hide for the first round. Bails if @eventstop reset state
// while this thread was mid-sleep in the countdown above.
OnFindGMIntro:
	if ( .active_event$ != "FindGM" ) end;
	hideoffnpc strnpcinfo(0);
	sleep 2000;
	npctalk "I will now teleport to random locations across different maps";
	sleep 2000;
	npctalk "You must find me to win the event!";
	sleep 2000;
	npctalk "This event will last for "+callfunc("F_InsertPlural",.findgm_rounds,"round");
	sleep 2000;
	npctalk "I can hide in: "+implode(.findgm_maps$,", ");
	sleep 2000;
	npctalk "Goodluck!";
	.findgm_round = 1;
	donpcevent strnpcinfo(0)+"::OnFindGMRoundStart";
	end;

OnFindGMRoundStart:
	// Bails if @eventstop reset state while this thread was mid-sleep.
	if ( .active_event$ != "FindGM" ) end;
	// Safety cutoff: replaces the old broken OnTimer1800000 approach
	// (which reset every round via initnpctimer and never actually
	// timed out). This checks real elapsed time instead.
	if ( gettimetick(2) - .findgm_startTick > .findgm_maxDuration ) {
		donpcevent strnpcinfo(0)+"::OnFindGMEnd";
		end;
	}
	.@mapIndex = rand(getarraysize(.findgm_maps$));
	.findgm_map$ = .findgm_maps$[.@mapIndex];

	do {
		.@x = rand(0,450);
		.@y = rand(0,450);
	} while (!checkcell(.findgm_map$,.@x,.@y,cell_chkpass));
	unitwarp getnpcid(0),.findgm_map$,.@x,.@y;
	announce "[Find "+.npcName$+"] : Round "+.findgm_round+"/"+.findgm_rounds+" - "+.npcName$+" is hiding somewhere in "+.findgm_map$+"! Go find him now!",bc_all|bc_yellow;
	delwaitingroom;
	waitingroom "Enter to claim prize!", 2, strnpcinfo(0)+"::OnFindGMRoundEnd",1;
	.findgm_status = 2;
	end;

OnFindGMRoundEnd:
	if(.findgm_status != 2) end;
	.findgm_status = 1;
	getwaitingroomusers strnpcinfo(0);
	for(.@i = 0; .@i < getarraysize(.@waitingroom_users); .@i++) {
		if(.@waitingroom_users[.@i] > 0) {
			.@winner = .@waitingroom_users[.@i];
		}
	}
	if(!attachrid(.@winner)) {
		.findgm_status = 2;
		end;
	}
	specialeffect2 EF_SPHERE;
	announce .npcName$+": I've been found by "+strcharinfo(0)+" in "+.findgm_map$+"! (Round "+.findgm_round+"/"+.findgm_rounds+")",bc_all|bc_yellow;
	if(.findgm_pointReward) {
		#EVENT_POINTS += .findgm_pointReward;
		dispbottom "You won "+.findgm_pointReward+" Event Point(s)! You now have "+#EVENT_POINTS+" Event Points.";
	}
	if(.findgm_zenyReward) {
		Zeny += .findgm_zenyAmount;
		dispbottom "You've gained "+.findgm_zenyAmount+" zeny for winning a round in the event.";
	}
	sleep 1000;
	if(.findgm_round >= .findgm_rounds) {
		donpcevent strnpcinfo(0)+"::OnFindGMEnd";
		end;
	}
	.findgm_round++;
	announce "[Find "+.npcName$+"] : Round "+.findgm_round+" starting in 5 seconds!",bc_all|bc_yellow;
	sleep 5000;
	donpcevent strnpcinfo(0)+"::OnFindGMRoundStart";
	end;

// Called by @eventstop via donpcevent "Dev Prime#findgm::OnFindGMForceReset".
// Runs on the walking unit itself: drop its waiting room, hide it,
// and pull it back to its parked spot regardless of what it was
// mid-doing when the reset happened.
OnFindGMForceReset:
	delwaitingroom;
	unitwarp getnpcid(0),"maintown",114,156;
	hideonnpc strnpcinfo(0);
	end;
}
