function	script	F_autobufftoggle	{
set @eac, eaclass();

if((@eac&EAJ_BASEMASK) != EAJ_ACOLYTE
	&& (@eac&EAJ_UPPERMASK) != EAJ_ALCHEMIST
	&& (@eac&EAJ_UPPERMASK) != EAJ_CRUSADER
	&& (@eac&EAJ_UPPERMASK) != EAJ_BARDDANCER
	) {
	mes "This feature is only allowed for acolyte, crusader, alchemist or bard/dancer based jobs.";
	close;
}

if(getcharid(1) == 0){
	mes "You need to be in a party to use AutoBuff.";
	close;
}

.@config_mode = 0;
if(getargcount() > 0)
	.@config_mode = getarg(0);

if(.@config_mode == 1)
	goto(config_menu);
goto(main_menu);
end;

main_menu:
    .@menu$ = "[Auto Buff Configuration]";
    mes .@menu$;

    if((@eac&EAJ_BASEMASK) != EAJ_ACOLYTE
        && (@eac&EAJ_UPPERMASK) != EAJ_ALCHEMIST
        && (@eac&EAJ_UPPERMASK) != EAJ_CRUSADER
        && (@eac&EAJ_UPPERMASK) != EAJ_BARDDANCER
        ) {
        mes "This feature is only allowed for acolyte, crusader, alchemist or bard/dancer based jobs.";
        close;
    }

    if(getcharid(1) == 0){
        mes "You need to be in a party to use the AutoBuff Device";
        close;
    }

    // State verification and cleanup inline
    .@current_state = ab_getstate();

    // Check for inconsistent state and force cleanup if needed
    if(.@current_state && !getstatus(SC_AUTOBUFF, 5)) {
        // State says ON but no actual buff status - force clear
        sc_end SC_AUTOBUFF;
        sleep2 100;
        .@current_state = 0; // Update local state
    }

    .@main_menu = select(.@current_state ? "[Stop]" : "[Start]", "[Configuration]", "[Cancel]");

        switch(.@main_menu){
        case 1:
            if(.@current_state) {
                sc_end SC_AUTOBUFF;
                sleep2 200; // Give time for cleanup
                goto(main_menu); // Return to menu after stopping
            } else {
                autobuff_start(86400); // Start AutoBuff directly, no item required
                close; // Auto-close after starting
            }
            break;
        case 2:
            goto(config_menu);
            break;
    }
    end;

			" ~ Return to save point",save_point_menu,
			" ~ Token of Siegfried",siegfried_menu,
			" ~ Priorize buff",priorizebuff_menu,
			" ~ Reset all",reset_all;

	// To disable an option, comment the line
	menu	"Return",main_menu,

			" ~ Buff Skills",buffskills_menu,

	end;

potion_pitcher_menu:

	.@menu$ = "[Potion Pitcher]";

	mes .@menu$;
	mes "--- Current configuration ---";
	mes "You can add / modify or remove the following skills";

	setarray .@lheal_skill[0],231;

	.@healskill_menu1$ = "";
	.@j = 0;

	// Check if the player have the skills
	for (.@i = 0; .@i < getskilllv(.@lheal_skill[0]); .@i++)
		.@lheal_skill2[.@i] = .@lheal_skill[0];

	if(getarraysize(.@lheal_skill2) == 0){
		clear;
		mes .@menu$;
		mes "Sorry, you don't have any kind of heal skills allowed";
		next;
		goto(regeneration_menu);
	}

	for (.@i = 0; .@i < getarraysize(.@lheal_skill2); .@i++)
		.@healskill_menu1$ = .@healskill_menu1$ + (.@i > 0 ? ":" : "") + ab_getautobuffstring(12,.@lheal_skill2[.@i],1,.@i);

	.@healskill_menu1$ = "[Cancel]:" + .@healskill_menu1$;
	.@i = select(.@healskill_menu1$);

	if(.@i == 1)
		goto(regeneration_menu);
	else
		.@i = .@i - 2;

	.@heal_skilllv = .@i + 1;

	if(.@skill_menu_state[.@heal_skilllv]){
		.@heal_menu = select("Modify","Disable","[Cancel]");
		switch(.@heal_menu){
			case 1 :
				do {
					clear;
					mes .@menu$;
					if(.@heal_skilllv == 5)
						mes "Choose the value of SP % to use the skill (ex : 10 to use skill if sp are below 10%)";
					else
						mes "Choose the value of HP % to use the skill (ex : 10 to use skill if hp are below 10%)";
					mes "Min value is 5, Max value is 90";
					input .@heal_minhpvalue;
				} while(.@heal_minhpvalue < 5 || .@heal_minhpvalue > 90);

				clear;
				mes .@menu$;
				mes "Choose your potion pitch configuration";
				.@heal_prio = select("Pitch Self Only","Pitch Party Leader Only","Pitch Specific Party Members","Pitch Everyone Except Self","Pitch Entire Party","[Cancel]");

						break;
					case 2:
						ab_setautobuff("14;1;"+ .@lheal_skill2[.@i] + ";" + .@heal_skilllv + ";" + .@heal_minhpvalue + ";1");

						do {
							clear;
							mes .@menu$;
							mes ab_getautobuffstring(12,0,.@lheal_skill2[.@i],.@heal_skilllv);

							if(getarraysize(.@player_heal_ids) == 0){
								mes .@menu$;
								mes "Sorry, you're the only player in the party";
								next;
								goto(potion_pitcher_menu);
							}

							.@k = select( .@player_heal_list$ ) - 1;

							if(.@player_heal_state[.@player_heal_ids[.@k]])
								ab_setautobuff("14;"+ .@lheal_skill2[.@i] + ";0;" + .@player_heal_ids[.@k] +";"+ .@heal_skilllv);

						break;
				}
				goto(potion_pitcher_menu);
				break;
			case 2:
				ab_setautobuff("14;0;"+ .@lheal_skill2[.@i] + ";" + .@heal_skilllv + ";0;0");
				goto(potion_pitcher_menu);
				break;

				break;
		}
	} else {

				.@heal_prio = select("Pitch Self Only","Pitch Party Leader Only","Pitch Specific Party Members","Pitch Everyone Except Self","Pitch Entire Party","[Cancel]");

				switch(.@heal_prio){
					case 1:
						ab_setautobuff("14;1;"+ .@lheal_skill2[.@i] + ";" + .@heal_skilllv + ";" + .@heal_minhpvalue + ";0");
						break;
					case 2:
						ab_setautobuff("14;1;"+ .@lheal_skill2[.@i] + ";" + .@heal_skilllv + ";" + .@heal_minhpvalue + ";1");
						break;
					case 3:
						ab_setautobuff("14;1;"+ .@lheal_skill2[.@i] + ";" + .@heal_skilllv + ";" + .@heal_minhpvalue + ";2");

						do {
							clear;
							mes .@menu$;
							mes ab_getautobuffstring(12,0,.@lheal_skill2[.@i],.@heal_skilllv);

							if(getarraysize(.@player_heal_ids) == 0){
								mes .@menu$;
								mes "Sorry, you're the only player in the party";
								next;
								goto(potion_pitcher_menu);
							}

							.@k = select( .@player_heal_list$ ) - 1;

							if(.@player_heal_state[.@player_heal_ids[.@k]])
								ab_setautobuff("14;"+ .@lheal_skill2[.@i] + ";0;" + .@player_heal_ids[.@k] +";"+ .@heal_skilllv);
							else
								ab_setautobuff("14;"+ .@lheal_skill2[.@i] + ";1;" + .@player_heal_ids[.@k] +";"+ .@heal_skilllv);

							.@l = select("Change another player","[Cancel]");
							if(.@l == 2)

						break;
				}
				goto(potion_pitcher_menu);
				break;
			case 2:

				break;
		}
	}

	goto(regeneration_menu);
	end;

	// Function to determine potion type based on item ID
	// Returns: 1=HP only, 2=SP only, 3=Both HP and SP
	function getPotionType {
		.@item_id = getarg(0);

		// HP-only potions (common healing items)
		setarray .@hp_potions[0], 12021, 12022, 12026, 12027, 501, 502, 503, 504, 506, 507, 508, 509, 510, 511, 512, 513, 514, 515, 516, 517, 518, 519, 520, 521, 522, 523, 525, 526, 545, 546, 547, 569, 570, 571, 572, 573, 574, 575, 11501;

		// SP-only potions (mana/sp restoration items)
		setarray .@sp_potions[0], 531, 505, 532, 533, 534, 535, 536, 537, 538, 539, 540, 541, 542, 543, 544, 11502;

		// Both HP and SP potions (special items like Yggdrasil)
		setarray .@both_potions[0], 607, 608, 526;

		// Check HP-only potions
		for(.@j = 0; .@j < getarraysize(.@hp_potions); .@j++) {
			if(.@hp_potions[.@j] == .@item_id) return 1;
		}

		// Check SP-only potions
		for(.@j = 0; .@j < getarraysize(.@sp_potions); .@j++) {
			if(.@sp_potions[.@j] == .@item_id) return 2;
		}

		// Check Both HP and SP potions
		for(.@j = 0; .@j < getarraysize(.@both_potions); .@j++) {
			if(.@both_potions[.@j] == .@item_id) return 1;
		}

		// Default to both if unknown (safer option)
		return 1;
	}

	// Get the potion type for the selected item
	.@potion_type = getPotionType(.@potion_menu_ids[.@i]);

	if(.@potion_menu_state[.@potion_menu_ids[.@i]]){
		.@potion_menu2 = select("Modify","Disable","[Cancel]");
		switch(.@potion_menu2){
			case 1:
				// Initialize values to 0
				.@potion_minhpvalue = 0;
				.@potion_minspvalue = 0;

				// Configure based on potion type
				if(.@potion_type == 1 || .@potion_type == 3) { // HP only or Both
					do {
						clear;
						mes .@menu$;
						mes "Choose the value of HP % to use the consumable (ex : 10 to use it if hp are below 10%)";
						mes "Min value is 5, Max value is 90";
						input .@potion_minhpvalue;
					} while(.@potion_minhpvalue < 5 || .@potion_minhpvalue > 90);
				}

				if(.@potion_type == 2 || .@potion_type == 3) { // SP only or Both
					do {
						clear;
						mes .@menu$;
						mes "Choose the value of SP % to use the consumable (ex : 10 to use it if sp are below 10%)";
						mes "Min value is 5, Max value is 90";
						input .@potion_minspvalue;
					} while(.@potion_minspvalue < 5 || .@potion_minspvalue > 90);
				}

				ab_setautobuff("1;1;"+ .@potion_menu_ids[.@i] + ";" + .@potion_minhpvalue + ";" + .@potion_minspvalue);
				goto(potion_menu);
				break;
			case 2:
				ab_setautobuff("1;0;"+ .@potion_menu_ids[.@i] + ";0;0");
				goto(potion_menu);
				break;
			case 3:
				goto(potion_menu);
				break;
		}
	}
	else{
		.@potion_menu2 = select("Add","[Cancel]");
		switch(.@potion_menu2){
			case 1:
				// Initialize values to 0
				.@potion_minhpvalue = 0;
				.@potion_minspvalue = 0;

				if(.@potion_type == 1 || .@potion_type == 3) { // HP only or Both

				//	ab_setautobuff("3;1;"+ .@skill_menu_ids[.@i] + ";" + .@buff_skilllv + ";0");
				//	goto(buffskills_menu);
				//} else
					.@buff_prio = select("[Cancel]","Buff Self Only","Buff Party Leader Only","Buff Specific Party Members","Buff Everyone Except Self","Buff Entire Party");

				switch(.@buff_prio){
					case 2:
						if(ab_skillinfo(1,.@skill_menu_ids[.@i])){
							mes "You can't select Self on this skill";
							next;
							goto(buffskills_menu);
						}
						ab_setautobuff("3;1;"+ .@skill_menu_ids[.@i] + ";" + .@buff_skilllv + ";0");
						break;
					case 3:
						ab_setautobuff("3;1;"+ .@skill_menu_ids[.@i] + ";" + .@buff_skilllv + ";1");
						break;
					case 4:
						ab_setautobuff("3;1;"+ .@skill_menu_ids[.@i] + ";" + .@buff_skilllv + ";2");

						do {
							clear;
							mes .@menu$;
							mes ab_getautobuffstring(3,.@skill_menu_ids[.@i], 1);

							if(getarraysize(.@player_buff_ids) == 0){
								mes .@menu$;
								mes "Sorry, you're the only player in the party";
								next;
								goto(buffskills_menu);
							}

							.@k = select( .@player_buff_list$ );

							if(.@player_buff_state[.@player_buff_ids[.@k]])
								ab_setautobuff("3;"+ .@skill_menu_ids[.@i] + ";0;" + .@player_buff_ids[.@k]);
							else
								ab_setautobuff("3;"+ .@skill_menu_ids[.@i] + ";1;" + .@player_buff_ids[.@k]);

							clear;
							mes ab_getautobuffstring(3,.@skill_menu_ids[.@i], 1);
							.@l = select("Change another player","[Cancel]");
							if(.@l == 2)
								goto(buffskills_menu);

						} while(1);

						goto(buffskills_menu);
						break;
					case 5:
						ab_setautobuff("3;1;"+ .@skill_menu_ids[.@i] + ";" + .@buff_skilllv + ";3");
						break;
					case 6:
						ab_setautobuff("3;1;"+ .@skill_menu_ids[.@i] + ";" + .@buff_skilllv + ";4");
						break;
				}
				goto(buffskills_menu);
				break;
			case 2:
				goto(config_menu);
				break;
		}
	}

	goto(config_menu);
	end;

buffitems_menu:
	clear;
	.@menu$ = "[Buff Items]";

	mes .@menu$;
	mes "--- Current configuration ---";
	setarray .@lbuff_item[0],645,656,657,12208,12210;
	setarray .@lbuff_sc[0],SC_ASPDPOTION0,SC_ASPDPOTION1,SC_ASPDPOTION2,SC_EXPBOOST,SC_ITEMBOOST;
	.@buffitems_menu1$ = "";

	for (.@i = 0; .@i < getarraysize(.@lbuff_item); .@i++)
		.@buffitems_menu1$ = .@buffitems_menu1$ + (.@i > 0 ? ":" : "") + ab_getautobuffstring(5,.@lbuff_item[.@i]);

	.@i = select(.@buffitems_menu1$) -1;

	.@buffitems_menu = select(.@buffitem_menu_state[.@lbuff_item[.@i]] ? "Disable" : "Enable","[Cancel]");
	if( .@buffitems_menu == 1 ){
		if(.@buffitem_menu_state[.@lbuff_item[.@i]])
			ab_setautobuff("5;0;"+ .@lbuff_item[.@i] + ";" + .@lbuff_sc[.@i]);
		else
			ab_setautobuff("5;1;"+ .@lbuff_item[.@i] + ";" + .@lbuff_sc[.@i]);
		goto(buffitems_menu);
	}
	else if( .@buffitems_menu == 2 ){
		ab_setautobuff("5;0;"+ .@lbuff_item[.@i] + ";0");
		goto(buffitems_menu);
	}

	goto(config_menu);
	end;

}

//
// ============================================================
//  @abstart / @abend atcommands
//  Alternative entry points into the F_autobufftoggle system.
//  Placed in an invisible, always-loaded NPC so bindatcmd fires
//  once on server start. Enforces the same job/party restrictions
//  as the item-based Start/Stop menu.
//
//  Players never need to own or interact with any item to use these
//  commands - they call the custom autobuff_start(<seconds>) source
//  command directly, which starts SC_AUTOBUFF with no item check.
// ============================================================
//
-	script	AutoBuffCommands	FAKE_NPC,{
OnInit:
	bindatcmd "abstart",strnpcinfo(3)+"::OnAbStart";
	bindatcmd "abend",strnpcinfo(3)+"::OnAbEnd";
	bindatcmd "abconfig",strnpcinfo(3)+"::OnAbConfig";
	end;

OnAbConfig:
	callfunc "F_autobufftoggle",1;
	end;

OnAbStart:
	callfunc "F_autobufftoggle";
	end;

OnAbEnd:
	.@current_state = ab_getstate();
	if(.@current_state && !getstatus(SC_AUTOBUFF, 5)){
		.@current_state = 0;
	}

	if(!.@current_state){
		dispbottom "AutoBuff is not currently running.";
		end;
	}

	sc_end SC_AUTOBUFF;
	sleep2 200;
	dispbottom "AutoBuff stopped.";
	end;
}
