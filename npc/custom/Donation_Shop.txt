//===== rAthena Script: Donation Shop (native shop UI + announce) =====
// Uses the ACTUAL client shop window (icon grid, quantity spinner) -
// not a text menu. This is done via an invisible "shop" NPC opened
// through callshop + npcshopattach. Because OnBuyItem is defined, it
// takes over the ENTIRE transaction from the engine: nothing is auto-
// charged or auto-given, we do both manually below. That's what lets
// us charge a custom item currency (Proof of Donation, 7179) instead
// of Zeny, and broadcast an announce per purchase.
//
// SELF-CONTAINED: no dependency on any other script file.
// ---------------------------------------------------------

// ---------------------------------------------------------
// Invisible backing shop (view -1 = never appears on the map,
// only reachable via callshop below). The prices here are
// COSMETIC ONLY - the client always labels them with a Zeny
// icon, but no Zeny is ever actually touched. The real currency
// and cost are enforced in OnBuyItem using the numbers in
// .cost[] further down, which are kept identical to these on
// purpose so the displayed price is accurate to the player.
//
// Index / Item                  / Price shown
//  0  Valkyrie Helm       (120134) : 20
//  1  Black Valkyrie Helm (120133) : 20
//  2  Cyan Valkyrie Helm  (120135) : 20
//  3  Gold Valkyrie Helm  (120136) : 20
//  4  Green Valkyrie Helm (120137) : 20
//  5  Red Valkyrie Helm   (120138) : 20
//  6  Cyclops Visor       (5135)   : 15
//  7  Black Dragon Wing   (120113) : 30
//  8  Dragon Wings        (120114) : 30
//  9  Cyan Dragon Wings   (120115) : 30
// 10  Gold Dragon Wings   (120116) : 30
// 11  Green Dragon Wings  (120117) : 30
// 12  Red Dragon Wings    (120118) : 30
// ---------------------------------------------------------
-	shop	DonationShop#hidden	-1,120134:20,120133:20,120135:20,120136:20,120137:20,120138:20,5135:15,120113:30,120114:30,120115:30,120116:30,120117:30,120118:30

prontera,155,166,4	script	Donation Shop#dshop	123,{

OnInteract:
	mes "^0000FF[ Donation Shop ]^000000";
	mes "^888888------------------------------^000000";
	mes "Trade your "+getitemname(.currency_id)+" for exclusive rewards.";
	mes "^888888------------------------------^000000";
	mes "You have: ^008000"+countitem(.currency_id)+"^000000 "+getitemname(.currency_id);
	mes "^888888------------------------------^000000";
	mes "^FF0000Note: prices shown use the Zeny icon, but you are charged "+getitemname(.currency_id)+", not Zeny.^000000";
	close2;

	npcshopattach "DonationShop#hidden";
	callshop "DonationShop#hidden",1;
	end;

// ===========================================================
// OnBuyItem - fires once per checkout, with the whole cart in
// @bought_nameid[]/@bought_quantity[]. Nothing is auto-charged
// or auto-given; we validate, charge, deliver, and announce.
// ===========================================================
OnBuyItem:
	.@count = getarraysize(@bought_nameid);
	if (.@count <= 0)
		end;

	// Validate the cart and total up the real cost in Proof of Donation
	.@total = 0;
	for (.@i = 0; .@i < .@count; .@i++) {
		.@qty = @bought_quantity[.@i];
		if (.@qty <= 0) {
			deletearray @bought_nameid[0], .@count;
			deletearray @bought_quantity[0], .@count;
			end;
		}

		.@found = false;
		for (.@j = 0; .@j < .item_count; .@j++) {
			if (.item_id[.@j] == @bought_nameid[.@i]) {
				.@total += .cost[.@j] * .@qty;
				.@found = true;
				break;
			}
		}
		if (!.@found) {
			deletearray @bought_nameid[0], .@count;
			deletearray @bought_quantity[0], .@count;
			message strcharinfo(0), "That item isn't available here.";
			end;
		}
	}

	if (countitem(.currency_id) < .@total) {
		deletearray @bought_nameid[0], .@count;
		deletearray @bought_quantity[0], .@count;
		message strcharinfo(0), "You don't have enough "+getitemname(.currency_id)+".";
		end;
	}

	// Charge once for the whole cart, then deliver + announce per item
	delitem .currency_id, .@total;
	for (.@i = 0; .@i < .@count; .@i++) {
		getitem @bought_nameid[.@i], @bought_quantity[.@i];
		announce strcharinfo(0)+" bought "+getitemname(@bought_nameid[.@i])+"!", bc_all, 0xFFCC00;
	}

	deletearray @bought_nameid[0], .@count;
	deletearray @bought_quantity[0], .@count;

	message strcharinfo(0), "Purchase complete! Spent "+.@total+" "+getitemname(.currency_id)+".";
	end;

OnInit:
	.currency_id = 7179; // Proof of Donation

	.item_count = 13;
	setarray .item_id[0],
		120134, 120133, 120135, 120136, 120137, 120138,
		5135,
		120113, 120114, 120115, 120116, 120117, 120118;

	setarray .cost[0],
		20, 20, 20, 20, 20, 20,
		15,
		30, 30, 30, 30, 30, 30;

	// Sanity check every ID resolves to a real item (catches
	// future typos/edits without failing silently).
	for (.@i = 0; .@i < .item_count; .@i++) {
		if (getitemname(.item_id[.@i]) == "null")
			debugmes "Donation Shop#dshop: item ID "+.item_id[.@i]+" (index "+.@i+") not found in item_db, skipping.";
	}

	end;

OnTouch:
	end;
}
